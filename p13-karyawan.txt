    private function ensureEmployee(): void
    {
        if (!auth()->check() || auth()->user()->role !== 'employee') {
            abort(response()->json([
                'status'  => false,
                'message' => 'Akses ditolak (khusus employee)',
            ], 403));
        }

    private function me(): User
    {
        return auth()->user();
    }

    private function companyOrFail()
    {
        $company = $this->me()->company;
        if (!$company) {
            abort(response()->json([
                'status'  => false,
                'message' => 'Company tidak ditemukan untuk user ini',
            ], 422));
        }
        return $company;
    }

    private function attendanceData(Attendance $a): array
    {
        return [
            'id'                    => (int) $a->id,
            'user_id'               => (int) $a->user_id,
            'company_id'            => (int) $a->company_id,
            'date'                  => $a->date?->toDateString(),
            'time_in'               => $a->time_in,
            'time_out'              => $a->time_out,
            'scheduled_in'          => $a->scheduled_in,
            'scheduled_out'         => $a->scheduled_out,
            'latlon_in'             => $a->latlon_in,
            'latlon_out'            => $a->latlon_out,
            'photo_in'              => $a->photo_in,
            'photo_out'             => $a->photo_out,
            'photo_in_url'          => $a->photo_in_url,
            'photo_out_url'         => $a->photo_out_url,
            'distance_in_m'         => $a->distance_in_m,
            'distance_out_m'        => $a->distance_out_m,
            'is_outside_radius_in'  => (bool) $a->is_outside_radius_in,
            'is_outside_radius_out' => (bool) $a->is_outside_radius_out,
            'status'                => $a->status,
            'late_minutes'          => (int) ($a->late_minutes ?? 0),
            'early_leave_minutes'   => (int) ($a->early_leave_minutes ?? 0),
            'overtime_minutes'      => (int) ($a->overtime_minutes ?? 0),
            'permission_id'         => $a->permission_id,
            'note'                  => $a->note,
            'marked_by'             => $a->marked_by,
        ];
    }

    private function permissionBrief(Permission $p): array
    {
        return [
            'id'         => $p->id,
            'type'       => $p->type,
            'type_label' => $p->type_label,
            'start_date' => $p->date_permission?->toDateString(),
            'end_date'   => $p->end_date?->toDateString(),
            'start_time' => $p->start_time,
            'end_time'   => $p->end_time,
            'reason'     => $p->reason,
        ];
    }

    /**
     * Absen yang masih terbuka (sudah masuk, belum pulang).
     * Mendukung shift malam: kalau hari ini belum ada, cek absen kemarin.
     */
    private function openAttendance(int $companyId, int $userId): ?Attendance
    {
        $today = Carbon::today()->toDateString();

        $att = Attendance::where('company_id', $companyId)
            ->where('user_id', $userId)
            ->whereDate('date', $today)
            ->first();

        if ($att) {
            return $att;
        }

        $yesterday = Attendance::where('company_id', $companyId)
            ->where('user_id', $userId)
            ->whereDate('date', Carbon::yesterday()->toDateString())
            ->whereNotNull('time_in')
            ->whereNull('time_out')
            ->first();

        // Hanya dianggap shift malam kalau jam pulang terjadwal < jam masuk
        if (
            $yesterday && $yesterday->scheduled_in && $yesterday->scheduled_out
            && $yesterday->scheduled_out < $yesterday->scheduled_in
        ) {
            return $yesterday;
        }

        return null;
    }

    // ----------------------------------------------------------
    // CHECK-IN
    // POST /api/company/employee/attendances/check-in
    // Body JSON: { latitude, longitude, photo (base64 string), note? }
    // ----------------------------------------------------------
    public function checkIn(Request $request): JsonResponse
    {
        $this->ensureEmployee();

        $request->validate([
            'latitude'  => 'required|numeric|between:-90,90',
            'longitude' => 'required|numeric|between:-180,180',
            'photo'     => 'required|string|min:100',
            'note'      => 'nullable|string|max:500',
        ], [
            'photo.required' => 'Foto absen wajib diambil.',
        ]);

        $user    = $this->me();
        $company = $this->companyOrFail();
        $today   = Carbon::today()->toDateString();
        $now     = Carbon::now();

        // Guard double check-in
        $existing = Attendance::where('company_id', $company->id)
            ->where('user_id', $user->id)
            ->whereDate('date', $today)
            ->first();

        if ($existing && $existing->time_in) {
            return response()->json([
                'status'  => false,
                'message' => 'Sudah check-in hari ini pada pukul ' . $existing->time_in,
            ], 422);
        }

        // Izin yang sudah di-approve untuk hari ini
        $permissions = $this->rules->approvedPermissionsOn($user->id, $company->id, $today);
        $dinasLuar   = $this->rules->findDinasLuar($permissions);

        // Lokasi
        $distanceM = $this->rules->distanceMeters($company, (float) $request->latitude, (float) $request->longitude);
        $radiusM   = $this->rules->radiusMeters($company);
        $outside   = $distanceM > $radiusM;

        if ($outside && !$dinasLuar) {
            return response()->json([
                'status'  => false,
                'message' => sprintf(
                    'Anda berada di luar radius kantor (%s m, batas %s m).',
                    number_format($distanceM, 0, ',', '.'),
                    number_format($radiusM, 0, ',', '.')
                ),
                'hint' => 'Jika sedang dinas luar, ajukan izin "Dinas luar" dan tunggu persetujuan HR.',
                'data' => [
                    'distance_m' => $distanceM,
                    'radius_m'   => $radiusM,
                ],
            ], 422);
        }

        // Shift & keterlambatan (memperhitungkan izin yang sudah di-approve)
        $shift        = $this->rules->resolveShift($user, $today);
        $scheduledIn  = $this->rules->scheduledIn($shift, $company, $today);
        $scheduledOut = $this->rules->scheduledOut($shift, $company, $today);
        $late         = $this->rules->calcLate($shift, $company, $now, $today, $permissions);
        $usedPerm     = $late['permission'] ?? $dinasLuar;

        // Simpan foto paling akhir, setelah semua validasi lolos
        $photoPath = $this->photos->storeBase64($request->photo);

        try {
            $attendance = Attendance::firstOrNew([
                'company_id' => $company->id,
                'user_id'    => $user->id,
                'date'       => $today,
            ]);

            $note = $this->rules->buildNote($usedPerm);
            if ($request->filled('note')) {
                $note = trim(($note ? $note . ' | ' : '') . 'Catatan: ' . $request->note);
            }

            $attendance->fill([
                'marked_by'            => null,
                'shift_id'             => $shift?->id,
                'scheduled_in'         => $scheduledIn?->format('H:i:s'),
                'scheduled_out'        => $scheduledOut?->format('H:i:s'),
                'time_in'              => $now->format('H:i:s'),
                'latlon_in'            => $request->latitude . ',' . $request->longitude,
                'photo_in'             => $photoPath,
                'distance_in_m'        => $distanceM,
                'is_outside_radius_in' => $outside,
                'status'               => $late['late_minutes'] > 0 ? 'late' : 'on_time',
                'late_minutes'         => $late['late_minutes'],
                'face_verified'        => false,
                'permission_id'        => $usedPerm?->id,
                'note'                 => $note,
            ]);
            $attendance->save();
        } catch (\Throwable $e) {
            $this->photos->delete($photoPath);
            throw $e;
        }

        $msg = 'Check-in berhasil';
        if ($dinasLuar && $outside) $msg .= ' (dinas luar)';
        if ($late['late_minutes'] > 0) $msg .= " (terlambat {$late['late_minutes']} menit)";
        elseif ($late['permission']) $msg .= ' (sesuai izin, tidak dihitung terlambat)';

        return response()->json([
            'status'  => true,
            'message' => $msg,
            'data'    => $this->attendanceData($attendance),
        ]);
    }

    // ----------------------------------------------------------
    // CHECK-OUT
    // POST /api/company/employee/attendances/check-out
    // Body JSON: { latitude, longitude, photo (base64 string), note? }
    // ----------------------------------------------------------
    public function checkOut(Request $request): JsonResponse
    {
        $this->ensureEmployee();

        $request->validate([
            'latitude'  => 'required|numeric|between:-90,90',
            'longitude' => 'required|numeric|between:-180,180',
            'photo'     => 'required|string|min:100',
            'note'      => 'nullable|string|max:500',
        ], [
            'photo.required' => 'Foto absen wajib diambil.',
        ]);

        $user    = $this->me();
        $company = $this->companyOrFail();
        $now     = Carbon::now();

        $attendance = $this->openAttendance($company->id, $user->id);

        if (!$attendance || !$attendance->time_in) {
            return response()->json([
                'status'  => false,
                'message' => 'Belum check-in hari ini',
            ], 422);
        }

        if ($attendance->time_out) {
            return response()->json([
                'status'  => false,
                'message' => 'Sudah check-out hari ini pada pukul ' . $attendance->time_out,
            ], 422);
        }

        $workDate    = $attendance->date->toDateString();
        $permissions = $this->rules->approvedPermissionsOn($user->id, $company->id, $workDate);
        $dinasLuar   = $this->rules->findDinasLuar($permissions);

        $distanceM = $this->rules->distanceMeters($company, (float) $request->latitude, (float) $request->longitude);
        $radiusM   = $this->rules->radiusMeters($company);
        $outside   = $distanceM > $radiusM;

        if ($outside && !$dinasLuar) {
            return response()->json([
                'status'  => false,
                'message' => sprintf(
                    'Anda berada di luar radius kantor (%s m, batas %s m).',
                    number_format($distanceM, 0, ',', '.'),
                    number_format($radiusM, 0, ',', '.')
                ),
                'hint' => 'Jika sedang dinas luar, ajukan izin "Dinas luar" dan tunggu persetujuan HR.',
                'data' => [
                    'distance_m' => $distanceM,
                    'radius_m'   => $radiusM,
                ],
            ], 422);
        }

        $shift = $attendance->shift_id ? Shift::find($attendance->shift_id) : null;
        $early = $this->rules->calcEarlyLeave($shift, $company, $now, $workDate, $permissions);

        $photoPath = $this->photos->storeBase64($request->photo);

        try {
            $attendance->time_out              = $now->format('H:i:s');
            $attendance->latlon_out            = $request->latitude . ',' . $request->longitude;
            $attendance->photo_out             = $photoPath;
            $attendance->distance_out_m        = $distanceM;
            $attendance->is_outside_radius_out = $outside;
            $attendance->early_leave_minutes   = $early['early_leave_minutes'];

            // Kalau saat check-in belum ada izin yang terpakai, pakai izin pulang cepat
            if (!$attendance->permission_id && $early['permission']) {
                $attendance->permission_id = $early['permission']->id;
                $attendance->note = trim(
                    ($attendance->note ? $attendance->note . ' | ' : '')
                        . $this->rules->buildNote($early['permission'])
                );
            }

            if ($request->filled('note')) {
                $attendance->note = trim(
                    ($attendance->note ? $attendance->note . ' | ' : '') . 'Catatan pulang: ' . $request->note
                );
            }

            $attendance->save();
        } catch (\Throwable $e) {
            $this->photos->delete($photoPath);
            throw $e;
        }

        $msg = 'Check-out berhasil';
        if ($early['early_leave_minutes'] > 0) $msg .= " (pulang {$early['early_leave_minutes']} menit lebih awal)";
        elseif ($early['permission']) $msg .= ' (sesuai izin pulang cepat)';

        return response()->json([
            'status'  => true,
            'message' => $msg,
            'data'    => $this->attendanceData($attendance),
        ]);
    }

        return response()->json([
            'status'      => true,
            'message'     => 'Status absensi hari ini',
            'checked_in'  => (bool) ($attendance && $attendance->time_in),

            'is_dinas_luar'            => (bool) $this->rules->findDinasLuar($permissions),
            'approved_permissions'     => $permissions->map(fn($p) => $this->permissionBrief($p))->values(),
            'pending_permissions'      => $pending->map(fn($p) => $this->permissionBrief($p))->values(),
            'data'        => $attendance ? $this->attendanceData($attendance) : null,
        ]);
    }

    // ----------------------------------------------------------
    // HISTORY
    // GET /api/company/employee/attendances/history
    // ----------------------------------------------------------
    public function history(Request $request): JsonResponse
    {
        $this->ensureEmployee();

        $user    = $this->me();
        $company = $this->companyOrFail();

        $limit = max(1, min((int) $request->query('limit', 30), 100));

        $attendances = Attendance::where('company_id', $company->id)
            ->where('user_id', $user->id)
            ->with(['shift', 'permission'])
            ->orderBy('date', 'desc')
            ->orderBy('id', 'desc')
            ->limit($limit)
            ->get();

        $data = $attendances->map(function ($a) {
            return array_merge($this->attendanceData($a), [
                'shift' => $a->shift ? [
                    'id'         => $a->shift->id,
                    'name'       => $a->shift->name,
                    'start_time' => $a->shift->start_time,
                    'end_time'   => $a->shift->end_time,
                ] : null,
                'permission' => $a->permission ? $this->permissionBrief($a->permission) : null,
            ]);
        })->values();

        return response()->json([
            'status'  => true,
            'message' => 'Riwayat absensi berhasil diambil',
            'data'    => $data,
        ]);
    }

    // ----------------------------------------------------------
    // SUMMARY STATISTIK BULANAN
    // GET /api/company/employee/stats/summary?month=&year=
    // ----------------------------------------------------------
    public function summary(Request $request): JsonResponse
    {
        $this->ensureEmployee();

        $user      = $this->me();
        $company   = $this->companyOrFail();
        $userId    = $user->id;
        $companyId = $company->id;

        $month = (int) $request->get('month', Carbon::now()->month);
        $year  = (int) $request->get('year', Carbon::now()->year);

        if ($month < 1 || $month > 12) {
            return response()->json([
                'success' => false,
                'message' => 'Bulan tidak valid (1-12).',
            ], 422);
        }

        $start = Carbon::createFromDate($year, $month, 1)->startOfMonth();
        $end   = $start->copy()->endOfMonth();

        $attendances = Attendance::where('company_id', $companyId)
            ->where('user_id', $userId)
            ->whereBetween('date', [$start->toDateString(), $end->toDateString()])
            ->get(['date', 'status', 'is_outside_radius_in']);

        $hadir     = $attendances->whereIn('status', ['on_time', 'overtime', 'guest'])->count();
        $terlambat = $attendances->where('status', 'late')->count();
        $alpha     = $attendances->where('status', 'absent')->count();
        $dinasLuar = $attendances->where('is_outside_radius_in', true)->count();

        $izin = Permission::where('user_id', $userId)
            ->where('company_id', $companyId)
            ->whereBetween('date_permission', [$start->toDateString(), $end->toDateString()])
            ->where('is_approved', true)
            ->count();

        $cuti = Leaves::where('user_id', $userId)
            ->where('company_id', $companyId)
            ->where('status', 'approved')
            ->where('start_date', '<=', $end->toDateString())
            ->where('end_date', '>=', $start->toDateString())
            ->count();

        $bulanLabel = [
            1 => 'Januari',
            2 => 'Februari',
            3 => 'Maret',
            4 => 'April',
            5 => 'Mei',
            6 => 'Juni',
            7 => 'Juli',
            8 => 'Agustus',
            9 => 'September',
            10 => 'Oktober',
            11 => 'November',
            12 => 'Desember',
        ];

        return response()->json([
            'success' => true,
            'message' => 'Ringkasan statistik kehadiran',
            'data'    => [
                'period' => [
                    'month' => $month,
                    'year'  => $year,
                    'label' => ($bulanLabel[$month] ?? $month) . ' ' . $year,
                ],
                'hadir'      => ['count' => $hadir,     'label' => 'Hari Hadir'],
                'terlambat'  => ['count' => $terlambat, 'label' => 'Terlambat'],
                'izin'       => ['count' => $izin,      'label' => 'Izin'],
                'cuti'       => ['count' => $cuti,      'label' => 'Cuti'],
                'alpha'      => ['count' => $alpha,     'label' => 'Alpha'],
                'dinas_luar' => ['count' => $dinasLuar, 'label' => 'Absen Dinas Luar'],
            ],
        ]);
    }

<?php

namespace App\Http\Controllers\Api\HrCompany;

use App\Http\Controllers\Controller;

use Illuminate\Support\Facades\DB;

class HrCompanyAttendanceController extends Controller
{


    // ============================================================
    // PRIVATE HELPERS
    // ============================================================

    private function ensureHr(): void
    {
        if (!auth()->check() || auth()->user()->role !== 'hr') {
            abort(response()->json([
                'status'  => false,
                'message' => 'Akses ditolak (khusus HR)',
            ], 403));
        }
    }

    private function companyOrFail(): Company
    {
        $companyId = auth()->user()->company_id ?? null;

        if (!$companyId) {
            abort(response()->json([
                'status'  => false,
                'message' => 'company_id tidak ditemukan pada user HR',
            ], 422));
        }

        $company = Company::find($companyId);

        if (!$company) {
            abort(response()->json([
                'status'  => false,
                'message' => 'Company tidak ditemukan',
            ], 404));
        }

        return $company;
    }

    private function haversineKm(float $lat1, float $lng1, float $lat2, float $lng2): float
    {
        $R    = 6371;
        $dLat = deg2rad($lat2 - $lat1);
        $dLng = deg2rad($lng2 - $lng1);
        $a    = sin($dLat / 2) ** 2
            + cos(deg2rad($lat1)) * cos(deg2rad($lat2)) * sin($dLng / 2) ** 2;

        return $R * 2 * atan2(sqrt($a), sqrt(1 - $a));
    }

    private function companyArray(Company $company): array
    {
        return [
            'id'        => (int)    $company->id,
            'name'      => (string) $company->name,
            'latitude'  => (string) $company->latitude,
            'longitude' => (string) $company->longitude,
            'radius_km' => (string) $company->radius_km,
            'time_in'   => (string) $company->time_in,
            'time_out'  => (string) $company->time_out,
        ];
    }

    private function attendanceArray(
        Attendance $a,
        User $employee,
        string $action,
        Company $company,
        ?float $distanceKm = null
    ): array {
        return [
            'attendance_id'        => (int) $a->id,
            'employee_id'          => (int) $employee->id,
            'employee_name'        => $employee->name,
            'employee_position'    => $employee->position,
            'employee_department'  => $employee->department,
            'action'               => $action,
            'date'                 => $a->date,
            'time_in'              => $a->time_in,
            'time_out'             => $a->time_out,
            'scheduled_in'         => $a->scheduled_in,
            'scheduled_out'        => $a->scheduled_out,
            'status'               => $a->status,
            'late_minutes'         => (int)  ($a->late_minutes        ?? 0),
            'early_leave_minutes'  => (int)  ($a->early_leave_minutes ?? 0),
            'overtime_minutes'     => (int)  ($a->overtime_minutes    ?? 0),
            'face_verified'        => (bool) ($a->face_verified       ?? false),
            'marked_by'            => $a->marked_by,
            'latlon_in'            => $a->latlon_in,
            'latlon_out'           => $a->latlon_out,
            'success_time'         => $action === 'checkin' ? $a->time_in : $a->time_out,
            'company'              => $this->companyArray($company),
            'distance_km'          => $distanceKm !== null ? round($distanceKm, 3) : null,
            'shift'                => $a->relationLoaded('shift') && $a->shift ? [
                'id'         => $a->shift->id,
                'name'       => $a->shift->name,
                'start_time' => $a->shift->start_time,
                'end_time'   => $a->shift->end_time,
            ] : null,
        ];
    }

    // ============================================================
    // SHIFT RESOLVER
    // Prioritas: user_shift_overrides → shift_group → default shift
    // ============================================================

    private function resolveShift(User $employee, string $date): ?Shift
    {
        // 1. Override spesifik user
        $override = DB::table('user_shift_overrides')
            ->where('user_id', $employee->id)
            ->where('status', 'active')
            ->where('start_date', '<=', $date)
            ->where(function ($q) use ($date) {
                $q->whereNull('end_date')->orWhere('end_date', '>=', $date);
            })
            ->whereNull('deleted_at')
            ->orderByDesc('start_date')
            ->first();

        if ($override) return Shift::find($override->shift_id);

        // 2. Group shift
        $groupIds = DB::table('shift_group_users')
            ->where('user_id', $employee->id)
            ->where(function ($q) use ($date) {
                $q->whereNull('start_date')->orWhere('start_date', '<=', $date);
            })
            ->where(function ($q) use ($date) {
                $q->whereNull('end_date')->orWhere('end_date', '>=', $date);
            })
            ->pluck('shift_group_id');

        if ($groupIds->isNotEmpty()) {
            $assignment = DB::table('shift_group_assignments')
                ->whereIn('shift_group_id', $groupIds)
                ->where('company_id', $employee->company_id)
                ->where('start_date', '<=', $date)
                ->where(function ($q) use ($date) {
                    $q->whereNull('end_date')->orWhere('end_date', '>=', $date);
                })
                ->whereNull('deleted_at')
                ->orderByDesc('start_date')
                ->first();

            if ($assignment) return Shift::find($assignment->shift_id);
        }

        // 3. Shift default company
        return Shift::where('company_id', $employee->company_id)
            ->where('is_default', true)
            ->first();
    }

    private function getScheduledIn(?Shift $shift, Company $company, string $date): ?Carbon
    {
        $time = $shift ? $shift->start_time : $company->time_in;
        return $time ? Carbon::parse($date . ' ' . $time) : null;
    }

    private function getScheduledOut(?Shift $shift, Company $company, string $date): ?Carbon
    {
        $time = $shift ? $shift->end_time : $company->time_out;
        return $time ? Carbon::parse($date . ' ' . $time) : null;
    }

    private function calcLateMinutes(?Shift $shift, Company $company, Carbon $actualIn, string $date): int
    {
        $scheduledIn = $this->getScheduledIn($shift, $company, $date);
        if (!$scheduledIn) return 0;

        $gracePeriod = $shift?->grace_period_minutes ?? 0;
        $deadline    = $scheduledIn->copy()->addMinutes($gracePeriod);

        if ($actualIn->lte($deadline)) return 0;

        return (int) $actualIn->diffInMinutes($scheduledIn);
    }

    private function calcEarlyLeaveMinutes(?Shift $shift, Company $company, Carbon $actualOut, string $date): int
    {
        $scheduledOut = $this->getScheduledOut($shift, $company, $date);
        if (!$scheduledOut) return 0;
        if ($actualOut->gte($scheduledOut)) return 0;

        return (int) $scheduledOut->diffInMinutes($actualOut);
    }

    // MARK EMPLOYEE ATTENDANCE
    // POST /api/company/hr/attendances/employees/mark
    // ============================================================
    public function markEmployeeAttendance(Request $request): JsonResponse
    {
        $this->ensureHr();
        $company = $this->companyOrFail();

        $validated = $request->validate([
            'employee_id' => 'required|exists:users,id',
            'action'      => 'required|in:checkin,checkout',
            'latitude'    => 'required|numeric|between:-90,90',
            'longitude'   => 'required|numeric|between:-180,180',
        ]);

        $employee = User::query()
            ->where('id', $validated['employee_id'])
            ->where('role', 'employee')
            ->where('company_id', $company->id)
            ->first();

        if (!$employee) {
            return response()->json([
                'status'  => false,
                'message' => 'User bukan employee atau bukan bagian company ini',
            ], 422);
        }

        if (empty($employee->face_embedding)) {
            return response()->json([
                'status'       => false,
                'face_warning' => true,
                'message'      => 'Karyawan ini belum melakukan registrasi wajah.',
                'hint'         => 'Silahkan minta karyawan untuk melakukan register face di device akun absensi mereka terlebih dahulu.',
            ], 422);
        }

        $distanceKm = $this->haversineKm(
            (float) $company->latitude,
            (float) $company->longitude,
            (float) $validated['latitude'],
            (float) $validated['longitude']
        );

        if ($distanceKm > (float) $company->radius_km) {
            return response()->json([
                'status'  => false,
                'message' => sprintf(
                    'Perangkat berada di luar radius attendance company (%s m dari batas %s m)',
                    number_format($distanceKm * 1000, 0),
                    number_format((float) $company->radius_km * 1000, 0)
                ),
                'data' => [
                    'distance_km' => round($distanceKm, 3),
                    'radius_km'   => (float) $company->radius_km,
                ],
            ], 422);
        }

        $today  = Carbon::today()->toDateString();
        $now    = Carbon::now();
        $latlon = $validated['latitude'] . ',' . $validated['longitude'];

        $shift        = $this->resolveShift($employee, $today);
        $scheduledIn  = $this->getScheduledIn($shift, $company, $today);
        $scheduledOut = $this->getScheduledOut($shift, $company, $today);

        $attendance = Attendance::firstOrNew([
            'company_id' => $company->id,
            'user_id'    => $employee->id,
            'date'       => $today,
        ]);

        $attendance->company_id    = $company->id;
        $attendance->user_id       = $employee->id;
        $attendance->date          = $today;
        $attendance->marked_by     = auth()->id();
        $attendance->shift_id      = $shift?->id;
        $attendance->scheduled_in  = $scheduledIn?->format('H:i:s');
        $attendance->scheduled_out = $scheduledOut?->format('H:i:s');
        $attendance->face_verified = true;

        if ($validated['action'] === 'checkin') {
            if ($attendance->time_in) {
                return response()->json([
                    'status'  => false,
                    'message' => 'Employee sudah check-in hari ini pada pukul ' . $attendance->time_in,
                ], 422);
            }

            $lateMinutes             = $this->calcLateMinutes($shift, $company, $now, $today);
            $attendance->time_in     = $now->format('H:i:s');
            $attendance->latlon_in   = $latlon;
            $attendance->status      = $lateMinutes > 0 ? 'late' : 'on_time';
            $attendance->late_minutes = $lateMinutes;
            $attendance->save();

            $msg = 'Check-in employee berhasil';
            if ($lateMinutes > 0) $msg .= " (terlambat {$lateMinutes} menit)";

            return response()->json([
                'status'  => true,
                'message' => $msg,
                'data'    => $this->attendanceArray($attendance->load('shift'), $employee, 'checkin', $company, $distanceKm),
            ]);
        }

        if (!$attendance->time_in) {
            return response()->json([
                'status'  => false,
                'message' => 'Employee belum check-in hari ini',
            ], 422);
        }

        if ($attendance->time_out) {
            return response()->json([
                'status'  => false,
                'message' => 'Employee sudah check-out hari ini pada pukul ' . $attendance->time_out,
            ], 422);
        }

        $earlyLeaveMinutes               = $this->calcEarlyLeaveMinutes($shift, $company, $now, $today);
        $attendance->time_out            = $now->format('H:i:s');
        $attendance->latlon_out          = $latlon;
        $attendance->early_leave_minutes = $earlyLeaveMinutes;
        $attendance->save();

        $msg = 'Check-out employee berhasil';
        if ($earlyLeaveMinutes > 0) $msg .= " (pulang {$earlyLeaveMinutes} menit lebih awal)";

        return response()->json([
            'status'  => true,
            'message' => $msg,
            'data'    => $this->attendanceArray($attendance->load('shift'), $employee, 'checkout', $company, $distanceKm),
        ]);
    }

<?php

class HrCompanyDailyReportController extends Controller
{
    private const USER_COLS = 'user:id,name,position,department,image_url';

    // ----------------------------------------------------------
    // HELPERS
    // ----------------------------------------------------------

    private function ensureHr(): void
    {
        if (!Auth::check() || Auth::user()->role !== 'hr') {
            abort(response()->json(['status' => false, 'message' => 'Akses ditolak (khusus HR)'], 403));
        }
    }

    private function companyId(): int
    {
        return (int) Auth::user()->company_id;
    }

    private function validationError($validator)
    {
        return response()->json([
            'status'  => false,
            'message' => $validator->errors()->first(),
            'errors'  => $validator->errors(),
        ], 422);
    }

    private function attendanceBrief(?Attendance $a): ?array
    {
        if (!$a) return null;

        return [
            'id'           => $a->id,
            'time_in'      => $a->time_in,
            'time_out'     => $a->time_out,
            'status'       => $a->status,
            'late_minutes' => (int) $a->late_minutes,
        ];
    }

    // ============================================================
    // GET /api/company/hr/daily-reports
    // filter: user_id, date, start, end, month, year, is_achieved,
    //         department, pending_evening=1, q, per_page
    // ============================================================
    public function index(Request $request)
    {
        $this->ensureHr();

        $query = DailyReport::with(self::USER_COLS)->where('company_id', $this->companyId());

        if ($request->filled('user_id')) $query->where('user_id', $request->user_id);
        if ($request->filled('date'))    $query->whereDate('date', $request->date);
        if ($request->filled('start') && $request->filled('end')) {
            $query->whereBetween('date', [$request->start, $request->end]);
        }
        if ($request->filled('month') && $request->filled('year')) {
            $query->whereMonth('date', $request->month)->whereYear('date', $request->year);
        }
        if ($request->filled('is_achieved')) {
            $query->where('is_achieved', filter_var($request->is_achieved, FILTER_VALIDATE_BOOLEAN))
                ->whereNotNull('achievement');
        }
        if ($request->filled('department')) {
            $query->whereHas('user', fn($q) => $q->where('department', $request->department));
        }
        if ($request->boolean('pending_evening')) {
            $query->whereNull('achievement');
        }
        if ($request->filled('q')) {
            $kw = $request->q;
            $query->whereHas('user', fn($q) => $q->where('name', 'like', "%{$kw}%"));
        }

        $reports = $query->orderByDesc('date')->orderBy('user_id')
            ->paginate((int) $request->get('per_page', 15));

        $reports->getCollection()->transform(fn($r) => $r->toApiArray());

        return response()->json([
            'status'  => true,
            'message' => 'Berhasil mengambil data daily report',
            'data'    => $reports,

    public function summary(Request $request)
    {
        $this->ensureHr();

            'month' => 'required|integer|between:1,12',
            'year'  => 'required|integer|min:2020|max:2099',

        if ($validator->fails()) {
            return $this->validationError($validator);
        }

        $rows = DailyReport::where('company_id', $this->companyId())
            ->whereMonth('date', $request->month)
            ->whereYear('date', $request->year)
            ->selectRaw('
                user_id,
                COUNT(*) as total_days,
                SUM(CASE WHEN is_achieved = 1 AND achievement IS NOT NULL THEN 1 ELSE 0 END) as total_achieved,
                SUM(CASE WHEN is_achieved = 0 AND achievement IS NOT NULL THEN 1 ELSE 0 END) as total_not_achieved,
                SUM(CASE WHEN achievement IS NULL THEN 1 ELSE 0 END) as total_pending
            ')
            ->groupBy('user_id')
            ->with(self::USER_COLS)
            ->get()
            ->map(function ($r) {
                $achieved    = (int) $r->total_achieved;
                $notAchieved = (int) $r->total_not_achieved;
                $filled      = $achieved + $notAchieved;

                return [
                    'user_id'            => (int) $r->user_id,
                    'user'               => $r->user ? [
                        'id'         => $r->user->id,
                        'name'       => $r->user->name,
                        'position'   => $r->user->position,
                        'department' => $r->user->department,
                        'image_url'  => $r->user->image_url ? asset($r->user->image_url) : null,
                    ] : null,
                    'total_days'         => (int) $r->total_days,
                    'total_achieved'     => $achieved,
                    'total_not_achieved' => $notAchieved,
                    'total_pending'      => (int) $r->total_pending,
                    'achievement_rate'   => $filled ? round($achieved / $filled * 100, 2) : 0,
                ];
            })
            ->sortByDesc('achievement_rate')
            ->values();

        return response()->json(['status' => true, 'data' => $rows]);

    {
        $this->ensureHr();

        $validator = Validator::make($request->all(), [
            'start_date' => 'required|date',
            'end_date'   => 'required|date|after_or_equal:start_date',
            'user_id'    => 'nullable|integer',
        ]);

        if ($validator->fails()) {
            return $this->validationError($validator);
        }

        $startDate = Carbon::parse($request->start_date)->toDateString();
        $endDate   = Carbon::parse($request->end_date)->toDateString();

        $query = DailyReport::where('company_id', $this->companyId())
            ->whereBetween('date', [$startDate, $endDate])
            ->with('user:id,name,position,department')
            ->orderBy('date')
            ->orderBy('user_id');

        if ($request->filled('user_id')) {
            $query->where('user_id', (int) $request->user_id);
        }

        $reports = $query->get();

        $summaryPerUser = $reports->groupBy('user_id')->map(function ($items) {
            $achieved    = $items->where('is_achieved', true)->whereNotNull('achievement')->count();
            $notAchieved = $items->where('is_achieved', false)->whereNotNull('achievement')->count();
            $pending     = $items->whereNull('achievement')->count();
            $filled      = $achieved + $notAchieved;

            return [
                'user'               => $items->first()->user,
                'total_days'         => $items->count(),
                'total_achieved'     => $achieved,
                'total_not_achieved' => $notAchieved,
                'total_pending'      => $pending,
                'achievement_rate'   => $filled ? round($achieved / $filled * 100, 1) : 0,
                'reports'            => $items,
            ];
        })->values();

        $periodLabel = Carbon::parse($startDate)->translatedFormat('d M Y')
            . ' - ' . Carbon::parse($endDate)->translatedFormat('d M Y');

        $fileName = 'daily-report-' . $startDate . '_to_' . $endDate
            . ($request->filled('user_id') ? '-user' . $request->user_id : '')
            . '.pdf';

        $pdf = \Barryvdh\DomPDF\Facade\Pdf::loadView('pdf.hr_daily_report', [
            'company'        => Auth::user()->company,
            'periodLabel'    => $periodLabel,
            'startDate'      => $startDate,
            'endDate'        => $endDate,
            'summaryPerUser' => $summaryPerUser,
            'totalReports'   => $reports->count(),
            'generatedAt'    => now()->format('d/m/Y H:i'),
        ])
            ->setPaper('a4', 'portrait')
            ->setOptions(['defaultFont' => 'sans-serif', 'isHtml5ParserEnabled' => true]);

        return $pdf->download($fileName);
    }

<?php

    // ----------------------------------------------------------
    // HELPERS
    // ----------------------------------------------------------

    private function ensureHr(): void
    {
        if (!auth()->check() || auth()->user()->role !== 'hr') {
            abort(response()->json([
                'status'  => false,
                'message' => 'Akses ditolak (khusus HR)',
            ], 403));
        }
    }

    private function companyId(): int
    {
        return (int) auth()->user()->company_id;
    }

    private function findOrFail($id): CompanyHoliday
    {
        return CompanyHoliday::query()
            ->where('company_id', $this->companyId())
            ->findOrFail($id);
    }

    /**
     * Kirim notifikasi SETELAH response dikirim ke HR,
     * supaya HR tidak menunggu proses kirim ke semua karyawan.
     * Error notifikasi tidak pernah menggagalkan simpan libur.
     */
    private function notifyLater(callable $fn): void
    {
        app()->terminating(function () use ($fn) {
            try {
                $fn();
            } catch (\Throwable $e) {
                Log::error('Gagal kirim notifikasi libur', ['error' => $e->getMessage()]);
            }
        });
    }

    private function rules(): array
    {
        return [
            'start_date' => ['required', 'date'],
            'end_date'   => ['required', 'date', 'after_or_equal:start_date'],
            'name'       => ['required', 'string', 'max:255'],
            'type'       => ['required', 'in:company,national'],
            'note'       => ['nullable', 'string', 'max:1000'],
            // false = simpan tanpa kirim notifikasi (default: kirim)
            'notify'     => ['nullable', 'boolean'],
        ];
    }

    private function overlapping(string $start, string $end, ?int $exceptId = null): ?CompanyHoliday
    {
        return CompanyHoliday::query()
            ->where('company_id', $this->companyId())
            ->when($exceptId, fn($q) => $q->where('id', '!=', $exceptId))
            ->whereDate('start_date', '<=', $end)
            ->whereDate('end_date', '>=', $start)
            ->first();
    }

    // ============================================================
    // GET /api/company/hr/holidays
    // filter: type, q, from, to, upcoming=1, year, per_page
    // ============================================================
    public function index(Request $request)
    {
        $this->ensureHr();

        $q = CompanyHoliday::query()->where('company_id', $this->companyId());

        if ($request->filled('type')) {
            $q->where('type', $request->type);
        }

        if ($request->filled('q')) {
            $search = trim($request->q);
            $q->where(function ($sub) use ($search) {
                $sub->where('name', 'like', "%{$search}%")
                    ->orWhere('note', 'like', "%{$search}%");
            });
        }

        if ($request->filled('from') && $request->filled('to')) {
            $q->whereDate('start_date', '<=', $request->to)
                ->whereDate('end_date', '>=', $request->from);
        } else {
            if ($request->filled('from')) $q->whereDate('end_date', '>=', $request->from);
            if ($request->filled('to'))   $q->whereDate('start_date', '<=', $request->to);
        }

        if ($request->filled('year')) {
            $q->whereYear('start_date', $request->year);
        }

        if ($request->boolean('upcoming')) {
            $q->upcoming()->orderBy('start_date');
        } else {
            $q->orderByDesc('start_date')->orderByDesc('end_date');
        }

        // Sebelumnya list HR mengirim data mentah (tanpa total_days dll),

    // PUT /api/company/hr/holidays/{id}
    // → kirim notifikasi "diperbarui" kalau tanggal / nama berubah
    // ============================================================
    public function update(Request $request, $id)
    {
        $this->ensureHr();

        $validated = $request->validate($this->rules(), [
            'end_date.after_or_equal' => 'Tanggal selesai tidak boleh sebelum tanggal mulai.',
        ]);

        $holiday = $this->findOrFail($id);

        if ($clash = $this->overlapping($validated['start_date'], $validated['end_date'], $holiday->id)) {
            return response()->json([
                'status'  => false,
                'message' => "Tanggal bentrok dengan libur \"{$clash->name}\" ({$clash->period_label}).",
            ], 422);
        }

        $oldStart = $holiday->start_date?->format('Y-m-d');
        $oldEnd   = $holiday->end_date?->format('Y-m-d');
        $oldName  = $holiday->name;

        $data = [
            'start_date' => $validated['start_date'],
            'end_date'   => $validated['end_date'],
            'name'       => $validated['name'],
            'type'       => $validated['type'],
            'note'       => $validated['note'] ?? null,
        ];

        $newStart = Carbon::parse($validated['start_date'])->format('Y-m-d');
        $newEnd   = Carbon::parse($validated['end_date'])->format('Y-m-d');

        // Tanggal mulai berubah → pengingat H-1 harus dikirim ulang
        if ($newStart !== $oldStart) {
            $data['reminder_sent_at'] = null;
        }

        $holiday->update($data);
        $holiday->refresh();

        $changed    = $newStart !== $oldStart || $newEnd !== $oldEnd || $validated['name'] !== $oldName;
        $willNotify = $changed && $request->boolean('notify', true) && $holiday->time_status !== 'past';

        if ($willNotify) {
            $this->notifyLater(fn() => $this->notifier->updated($holiday));
        }

        return response()->json([
            'status'       => true,
            'message'      => 'Libur berhasil diperbarui' . ($willNotify ? ', notifikasi dikirim ke karyawan' : ''),
            'notification' => $willNotify ? 'queued' : 'skipped',
            'data'         => $holiday->toApiArray(),
        ]);
    }

    // DELETE /api/company/hr/holidays/{id}
    // → kirim notifikasi "dibatalkan" kalau libur belum lewat
    // ============================================================
    public function destroy(Request $request, $id)
    {
        $this->ensureHr();

        $holiday = $this->findOrFail($id);

        $willNotify = $request->boolean('notify', true) && $holiday->time_status !== 'past';

        // Salinan untuk isi notifikasi (model asli sudah terhapus saat notifikasi jalan)
        $snapshot = (new CompanyHoliday())->forceFill($holiday->getAttributes());
        $snapshot->id = $holiday->id;

        $holiday->delete();

        if ($willNotify) {
            $this->notifyLater(fn() => $this->notifier->canceled($snapshot));
        }

        return response()->json([
            'status'       => true,
            'message'      => 'Libur berhasil dihapus' . ($willNotify ? ', notifikasi dikirim ke karyawan' : ''),
            'notification' => $willNotify ? 'queued' : 'skipped',
        ]);
    }

    // ============================================================
    // POST /api/company/hr/holidays/{id}/notify   (BARU)
    // Kirim ulang notifikasi libur secara manual — langsung, bukan antre.
    // Response berisi jumlah terkirim / gagal, berguna untuk tes.
    // ============================================================
    public function notify($id)
    {
        $this->ensureHr();

        $holiday = $this->findOrFail($id);

        if ($holiday->time_status === 'past') {
            return response()->json([
                'status'  => false,
                'message' => 'Libur ini sudah lewat.',
            ], 422);

        try {
            $result = $this->notifier->created($holiday);
        } catch (\Throwable $e) {
            return response()->json([
                'status'  => false,
                'message' => 'Gagal kirim notifikasi: ' . $e->getMessage(),
            ], 500);
        }

        return response()->json([
            'status'  => true,
            'message' => "Notifikasi terkirim ke {$result['sent']} karyawan",
            'data'    => $result,
        ]);
    }

    // POST /api/company/hr/leaves/{id}/approve
    // body (opsional): { approval_note }

    public function approve(Request $request, int $id)
    {
        $this->ensureHr();
        $request->validate(['approval_note' => ['nullable', 'string', 'max:500']]);

        $hr = $request->user();

        $leave = DB::transaction(function () use ($id, $hr, $request) {
            $leave = Leaves::query()
                ->lockForUpdate()
                ->where('company_id', $this->companyId())
                ->findOrFail($id);

            if ($leave->status !== 'pending') {
                abort(response()->json([
                    'status'  => false,
                    'message' => 'Hanya cuti pending yang bisa di-approve.',
                ], 422));
            }

            $leave->update([
                'status'        => 'approved',
                'approved_by'   => $hr->id,
                'approved_at'   => now(),
                'approval_note' => $request->approval_note,
            ]);

            return $leave;
        });

        // 🔔 Notifikasi ke karyawan
        $this->approvals->decided(ApprovalNotifier::KIND_LEAVE, $leave, true, $request->approval_note);

        return response()->json([
            'status'  => true,
            'message' => 'Cuti disetujui',
            'data'    => $leave->fresh([self::USER_COLUMNS, 'approver:id,name'])->toApiArray(),
        ]);
    }

            return response()->json([
                'success' => false,
                'message' => 'Validasi gagal',
                'errors'  => $validator->errors(),

        }

        // Pastikan employee milik company yang sama
        $employee = User::where('id', $request->user_id)
            ->where('company_id', Auth::user()->company_id)
            ->first();

        if (!$employee) {
            return response()->json([
                'success' => false,
                'message' => 'Karyawan tidak ditemukan.',

            ? $request->monthly_installment
            : round($request->amount / $installments, 2);

            'payment_date' => 'required|date',
            'method'       => 'required|in:manual,lump_sum',
            'note'         => 'nullable|string|max:500',
        ]);

        if ($validator->fails()) {
            return response()->json([
                'success' => false,
                'errors'  => $validator->errors(),
            ], 422);
        }

        // Tidak boleh bayar melebihi sisa hutang
        $amountPaid   = min((float) $request->amount_paid, (float) $loan->balance);
        $balanceAfter = round((float) $loan->balance - $amountPaid, 2);

        $payment = LoanPayment::create([
            'loan_id'         => $loan->id,
            'user_id'         => $loan->user_id,
            'company_id'      => $loan->company_id,
            'amount_expected' => $loan->monthly_installment,
            'amount_paid'     => $amountPaid,
            'balance_after'   => $balanceAfter,
            'payment_date'    => $request->payment_date,
            'method'          => $request->method,
            'payroll_id'      => null, // akan diisi otomatis nanti saat modul payroll aktif
            'note'            => $request->note,
            'recorded_by'     => Auth::id(),
        ]);

        // Update balance & cek lunas
        $newStatus = $balanceAfter <= 0 ? 'paid' : 'active';
        $loan->update([
            'balance' => $balanceAfter,
            'status'  => $newStatus,
        ]);

        return response()->json([
            'success' => true,
            'message' => $newStatus === 'paid'
                ? 'Pembayaran berhasil. Pinjaman sudah LUNAS!'
                : 'Pembayaran berhasil dicatat.',
            'data'    => [
                'payment'       => $payment->load('recordedBy:id,name'),
                'loan_status'   => $newStatus,
                'balance_after' => $balanceAfter,
            ],
        ], 201);
    }

     * HR hapus record pembayaran jika salah input

    public function deletePayment($id, $paymentId)
    {
        $loan = Loan::where('company_id', Auth::user()->company_id)->findOrFail($id);

        $payment = LoanPayment::where('loan_id', $id)->findOrFail($paymentId);

        $restoredBalance = round((float) $loan->balance + (float) $payment->amount_paid, 2);

        $loan->update([
            'balance' => $restoredBalance,
            // Jika sebelumnya paid karena pembayaran ini, kembalikan ke active
            'status'  => $loan->status === 'paid' ? 'active' : $loan->status,
        ]);

            'message' => 'Record pembayaran berhasil dihapus. Balance telah dikembalikan.',
            'data'    => [

            'type'               => ['sometimes', 'required', Rule::in(Note::TYPES)],
            'image_url'          => 'nullable|image|mimes:jpg,jpeg,png|max:2048',
            'remove_image'       => 'nullable|boolean',
            'target_achievement' => 'nullable|string|max:2000',
            'reason'             => 'nullable|string|max:2000',
        ], self::MESSAGES);

        if ($validator->fails()) {
            return $this->validationError($validator);
        }

        // Sebelumnya: image_url tanpa file ikut masuk validated() sebagai null
        // → gambar lama terhapus dari database tapi file-nya tertinggal.
        $data = collect($validator->validated())->except(['image_url', 'remove_image'])->all();

        if ($request->hasFile('image_url')) {
            $this->deleteImage($note->image_url);
            $data['image_url'] = $this->uploadImage($request->file('image_url'));
        } elseif ($request->boolean('remove_image')) {
            $this->deleteImage($note->image_url);
            $data['image_url'] = null;
        }

        // Isi berubah → tandai belum dibaca lagi supaya karyawan tahu ada perubahan
        if (array_intersect_key($data, array_flip(['title', 'note', 'type']))) {
            $data['is_read'] = false;
            $data['read_at'] = null;
        }

        $note->update($data);

        return response()->json([
            'status'  => true,
            'message' => 'Catatan berhasil diupdate',
            'data'    => $note->fresh($this->relations())->toApiArray(),
        ]);
    }

            'year'  => 'nullable|integer|min:2020|max:2099',
            'type'  => ['nullable', Rule::in(Note::TYPES)],
        ]);

        if ($validator->fails()) {
            return $this->validationError($validator);
        }

        $query = Note::with(['user:id,name,position,department', 'creator:id,name'])
            ->where('company_id', $this->companyId());

        if ($request->filled('type'))    $query->where('type', $request->type);
        if ($request->filled('user_id')) $query->where('user_id', $request->user_id);
        if ($request->filled('month') && $request->filled('year')) {
            $query->whereMonth('created_at', $request->month)
                ->whereYear('created_at', $request->year);
        } elseif ($request->filled('year')) {
            $query->whereYear('created_at', $request->year);
        }

        $notes = $query->orderByDesc('created_at')->get();

        $stats = [
            'total'       => $notes->count(),
            'warning'     => $notes->where('type', 'warning')->count(),
            'praise'      => $notes->where('type', 'praise')->count(),

            'general'     => $notes->where('type', 'general')->count(),
            'unread'      => $notes->where('is_read', false)->count(),

        $month       = $request->month ? (int) $request->month : null;
        $year        = $request->year ?? now()->year;
        $periodLabel = $month

            ->setPaper('a4', 'portrait')
            ->setOptions(['defaultFont' => 'sans-serif', 'isHtml5ParserEnabled' => true]);

        return $pdf->download('notes-' . now()->format('Y-m-d') . '.pdf');
    }

    // POST /api/company/hr/overtimes/{id}/approve

    public function approve(Request $request, int $id)
    {
        $this->ensureHr();

        return DB::transaction(function () use ($hr, $request, $id) {
            $overtime = OvertimeRequest::query()
                ->lockForUpdate()
                ->where('company_id', $hr->company_id)
                ->findOrFail($id);

            if ($overtime->status !== 'pending') {
                return response()->json([
                    'status'  => false,
                    'message' => 'Hanya request pending yang bisa di-approve.',
                ], 422);
            }

            $attendance = $this->findAttendance($overtime);

            $overtime->update([
                'status'        => 'approved',
                'approved_by'   => $hr->id,
                'approved_at'   => now(),
                'approval_note' => $request->approval_note,
                'attendance_id' => $overtime->attendance_id ?? $attendance?->id,
            ]);

            // Sebelumnya: status absen ditimpa jadi 'overtime' dan marked_by = HR.
            // Akibatnya status 'late' hilang dari rekap & absen dianggap input manual.
            // Sekarang hanya kolom lembur yang diubah.
            if ($attendance) {
                $attendance->update([
                    'overtime_minutes'  => (int) $overtime->minutes,
                    'approved_overtime' => true,
                ]);
            }

            // 🔔 Notifikasi ke karyawan
            $this->approvals->decided(ApprovalNotifier::KIND_OVERTIME, $overtime, true, $request->approval_note);

            $overtime->load([self::USER_COLUMNS, 'attendance', 'approver:id,name']);

            return response()->json([
                'status'  => true,
                'message' => 'Lembur disetujui',
                'data'    => array_merge($this->format($overtime, true), [
                    'attendance_updated' => (bool) $attendance,
                ]),
            ]);
        });
    }

<?php

namespace App\Http\Controllers\Api\HrCompany;

use App\Http\Controllers\Controller;
use App\Models\PayrollComponent;
use App\Models\Payrool;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Validation\Rule;

class HrCompanyPayrollComponentController extends Controller
{
    private function resolvePayroll(int $payrollId): Payrool
    {
        $hr = Auth::user();

        if (!in_array($hr->role, ['hr', 'superadmin'])) {
            abort(403, 'Akses ditolak. Hanya HR atau Superadmin yang dapat mengakses fitur ini.');
        }

        return Payrool::whereHas('user', fn($q) => $q->where('company_id', $hr->company_id))
            ->findOrFail($payrollId);
    }

    private function recalculateNetPay(Payrool $payroll): void
    {
        $totalAddition  = $payroll->components()->where('type', 'addition')->sum('amount');
        $totalDeduction = $payroll->components()->where('type', 'deduction')->sum('amount');

        $net = $payroll->base_salary
            + $payroll->overtime_pay
            + $payroll->bonus
            + $totalAddition
            - $totalDeduction;

        $payroll->update([
            'allowance'  => $totalAddition,
            'deductions' => $totalDeduction,
            'net_pay'    => max(0, $net),
        ]);
    }

    // =========================================================================
    // INDEX
    // =========================================================================
    public function index(int $payrollId): JsonResponse
    {
        $payroll    = $this->resolvePayroll($payrollId);
        $components = $payroll->components()->orderBy('type')->orderBy('name')->get();

        $additions  = $components->where('type', 'addition')->values();
        $deductions = $components->where('type', 'deduction')->values();

        return response()->json([
            'success' => true,
            'message' => 'Komponen payroll berhasil diambil.',
            'data'    => [
                'payroll_id'      => $payroll->id,
                'additions'       => $additions,
                'deductions'      => $deductions,
                'total_addition'  => $additions->sum('amount'),
                'total_deduction' => $deductions->sum('amount'),
                'current_net_pay' => $payroll->net_pay,
            ],
        ]);
    }

    // =========================================================================
    // STORE (satu komponen)
    // =========================================================================
    public function store(Request $request, int $payrollId): JsonResponse
    {
        $payroll = $this->resolvePayroll($payrollId);

        if ($payroll->status !== 'draft') {
            return response()->json([
                'success' => false,
                'message' => 'Komponen hanya bisa ditambahkan pada payroll berstatus draft.',
            ], 422);
        }

        $validated = $request->validate([
            'name'   => 'required|string|max:100',
            'type'   => ['required', Rule::in(['addition', 'deduction'])],
            'amount' => 'required|numeric|min:0',
            'note'   => 'nullable|string|max:255',
        ]);

        $component = $payroll->components()->create($validated);
        $this->recalculateNetPay($payroll->fresh());

        return response()->json([
            'success' => true,
            'message' => 'Komponen berhasil ditambahkan.',
            'data'    => [
                'component'   => $component,
                'new_net_pay' => $payroll->fresh()->net_pay,
            ],
        ], 201);
    }

    // =========================================================================
    // STORE BULK (banyak komponen sekaligus)
    // =========================================================================
    public function storeBulk(Request $request, int $payrollId): JsonResponse
    {
        $payroll = $this->resolvePayroll($payrollId);

        if ($payroll->status !== 'draft') {
            return response()->json([
                'success' => false,
                'message' => 'Komponen hanya bisa ditambahkan pada payroll berstatus draft.',
            ], 422);
        }

        $validated = $request->validate([
            'components'          => 'required|array|min:1|max:50',
            'components.*.name'   => 'required|string|max:100',
            'components.*.type'   => ['required', Rule::in(['addition', 'deduction'])],
            'components.*.amount' => 'required|numeric|min:0',
            'components.*.note'   => 'nullable|string|max:255',
        ]);

        $now        = now();
        $insertData = array_map(function ($item) use ($payroll, $now) {
            return [
                'payroll_id' => $payroll->id,
                'name'       => $item['name'],
                'type'       => $item['type'],
                'amount'     => $item['amount'],
                'note'       => $item['note'] ?? null,
                'created_at' => $now,
                'updated_at' => $now,
            ];
        }, $validated['components']);

        PayrollComponent::insert($insertData);
        $this->recalculateNetPay($payroll->fresh());

        return response()->json([
            'success' => true,
            'message' => count($insertData) . ' komponen berhasil ditambahkan.',
            'data'    => [
                'total_inserted' => count($insertData),
                'new_net_pay'    => $payroll->fresh()->net_pay,
            ],
        ], 201);
    }

    // =========================================================================
    // UPDATE (satu komponen)
    // =========================================================================
    public function update(Request $request, int $payrollId, int $componentId): JsonResponse
    {
        $payroll = $this->resolvePayroll($payrollId);

        if ($payroll->status !== 'draft') {
            return response()->json([
                'success' => false,
                'message' => 'Komponen hanya bisa diubah pada payroll berstatus draft.',
            ], 422);
        }

        $component = PayrollComponent::where('payroll_id', $payrollId)->findOrFail($componentId);

        $validated = $request->validate([

    public function generate(Request $request): JsonResponse
    {
        $hr = $this->authorizeHR();

        $validated = $request->validate([
            'period_start'             => 'required|date',
            'period_end'               => 'required|date|after_or_equal:period_start',
            'overtime_rate_per_minute' => 'nullable|numeric|min:0',
        ]);

        $overtimeRate = $validated['overtime_rate_per_minute'] ?? 0;

        $employees = User::where('company_id', $hr->company_id)
            ->whereNotIn('role', ['hr', 'superadmin', 'COMPANY'])
            ->get();

        $generated = [];
        $skipped   = [];

        DB::transaction(function () use ($employees, $validated, $overtimeRate, &$generated, &$skipped) {
            foreach ($employees as $employee) {
                $exists = Payrool::where('user_id', $employee->id)
                    ->where('period_start', $validated['period_start'])
                    ->where('period_end', $validated['period_end'])
                    ->exists();

                if ($exists) {
                    $skipped[] = [
                        'id'     => $employee->id,
                        'name'   => $employee->name,
                        'reason' => 'Sudah ada payroll untuk periode ini',
                    ];
                    continue;
                }

                $baseSalary = $employee->salary ?? 0;

                $overtimeMinutes = OvertimeRequest::where('user_id', $employee->id)
                    ->where('status', 'approved')
                    ->whereBetween('date', [$validated['period_start'], $validated['period_end']])
                    ->sum('minutes');

                $overtimePay = $overtimeMinutes * $overtimeRate;

                $loanDeduction = Loan::where('user_id', $employee->id)
                    ->where('status', 'approved')
                    ->where('balance', '>', 0)
                    ->get()
                    ->sum(fn($loan) => $loan->installments > 0 ? ($loan->amount / $loan->installments) : 0);

                $net = max(0, $baseSalary + $overtimePay - $loanDeduction);

                $payroll = Payrool::create([
                    'user_id'      => $employee->id,
                    'period_start' => $validated['period_start'],
                    'period_end'   => $validated['period_end'],
                    'base_salary'  => $baseSalary,
                    'allowance'    => 0,
                    'deductions'   => $loanDeduction,
                    'overtime_pay' => $overtimePay,
                    'bonus'        => 0,
                    'net_pay'      => $net,
                    'status'       => 'draft',
                ]);

                $generated[] = [
                    'payroll_id'   => $payroll->id,
                    'user_id'      => $employee->id,
                    'name'         => $employee->name,
                    'base_salary'  => $baseSalary,
                    'overtime_pay' => $overtimePay,
                    'deductions'   => $loanDeduction,
                    'net_pay'      => $net,
                ];
            }
        });

        return response()->json([
            'success'   => true,
            'message'   => 'Generate payroll selesai.',
            'generated' => $generated,
            'skipped'   => $skipped,
            'summary'   => [
                'total_generated' => count($generated),
                'total_skipped'   => count($skipped),
                'total_net_pay'   => collect($generated)->sum('net_pay'),
            ],
        ], 201);
    }

    // =========================================================================
    // SLIP
    // =========================================================================
    public function slip(int $id): JsonResponse
    {
        $hr = $this->authorizeHR();

        $payroll = Payrool::with([
            'user:id,name,email,position,department,salary,company_id,phone,image_url',
            'components' => fn($q) => $q->orderBy('type')->orderBy('name'),
        ])
            ->whereHas('user', fn($q) => $q->where('company_id', $hr->company_id))
            ->findOrFail($id);

        $workingDays = $this->countWorkingDays($payroll->period_start, $payroll->period_end);

        $presentDays = Attendance::where('user_id', $payroll->user_id)
            ->whereBetween('date', [$payroll->period_start, $payroll->period_end])
            ->whereIn('status', ['on_time', 'late', 'overtime'])
            ->count();

        $additions  = $payroll->components->where('type', 'addition')->values();
        $deductions = $payroll->components->where('type', 'deduction')->values();

        return response()->json([
            'success' => true,
            'message' => 'Data slip gaji berhasil diambil.',
            'data'    => [
                'payroll'         => $payroll->makeHidden('components'),
                'additions'       => $additions,
                'deductions'      => $deductions,
                'total_addition'  => $additions->sum('amount'),
                'total_deduction' => $deductions->sum('amount'),
                'attendance'      => [
                    'working_days' => $workingDays,
                    'present_days' => $presentDays,
                    'absent_days'  => max(0, $workingDays - $presentDays),
                ],
            ],
        ]);
    }

    // =========================================================================
    // PRIVATE HELPER
    // =========================================================================
    private function countWorkingDays(string $start, string $end): int
    {
        $startDate = Carbon::parse($start);
        $endDate   = Carbon::parse($end);
        $days      = 0;

        while ($startDate->lte($endDate)) {
            if ($startDate->isWeekday()) {
                $days++;
            }
            $startDate->addDay();
        }

        return $days;
    }

use App\Services\Notifiers\ApprovalNotifier;
use Illuminate\Http\Request;

class HrCompanyPermissionController extends Controller
{
    public function __construct(
        private AttendanceRuleService $rules,
        private ApprovalNotifier $approvals,
    ) {}

    // Kolom user yang aman dikirim. JANGAN kirim seluruh user:
    // face_embedding bisa puluhan KB dan ikut terkirim di setiap baris.
    private const USER_COLUMNS = 'user:id,name,email,phone,position,department,image_url';

    private function ensureHr(): void
    {
        if (!auth()->check() || auth()->user()->role !== 'hr') {
            abort(response()->json(['status' => false, 'message' => 'Akses ditolak (khusus HR)'], 403));
        }
    }

    private function companyId(): int
    {
        $companyId = auth()->user()->company_id ?? null;
        if (!$companyId) {
            abort(response()->json(['status' => false, 'message' => 'Company ID tidak ditemukan'], 422));
        }
        return (int) $companyId;
    }

    private function findOrFail($id): Permission
    {
        return Permission::query()
            ->with([self::USER_COLUMNS, 'approver:id,name'])
            ->where('company_id', $this->companyId())
            ->findOrFail($id);
    }

    private function applyFilters($q, Request $request)
    {
        if ($request->filled('status')) {
            match ($request->status) {
                'pending'  => $q->whereNull('is_approved'),
                'approved' => $q->where('is_approved', true),
                'rejected' => $q->where('is_approved', false),
                default    => null,
            };
        }

        if ($request->filled('type')) {
            $q->where('type', $request->type);
        }

        if ($request->filled('user_id')) {
            $q->where('user_id', $request->user_id);
        }

        if ($request->filled('q')) {
            $keyword = $request->q;
            $q->whereHas('user', function ($u) use ($keyword) {
                $u->where(function ($w) use ($keyword) {
                    $w->where('name', 'like', "%{$keyword}%")
                        ->orWhere('email', 'like', "%{$keyword}%");
                });
            });
        }

        // Filter berdasarkan TANGGAL IZIN, bukan tanggal pengajuan
        if ($request->filled('date_from')) {
            $q->whereDate('date_permission', '>=', $request->date_from);
        }
        if ($request->filled('date_to')) {
            $q->whereDate('date_permission', '<=', $request->date_to);
        }
        if ($request->filled('month') && $request->filled('year')) {
            $q->whereMonth('date_permission', $request->month)
                ->whereYear('date_permission', $request->year);
        } elseif ($request->filled('year')) {
            $q->whereYear('date_permission', $request->year);
        }

        return $q;
    }

    private function format(Permission $p): array
    {
        return [
            'id'              => $p->id,
            'type'            => $p->type,
            'type_label'      => $p->type_label,
            'date_permission' => $p->date_permission?->toDateString(),
            'end_date'        => $p->end_date?->toDateString(),
            'total_days'      => $p->date_permission
                ? $p->date_permission->diffInDays($p->end_date ?? $p->date_permission, true) + 1
                : 1,
            'start_time'      => $p->start_time ? substr($p->start_time, 0, 5) : null,
            'end_time'        => $p->end_time ? substr($p->end_time, 0, 5) : null,
            'reason'          => $p->reason,
            'image'           => $p->image,
            'image_url'       => $p->image ? asset($p->image) : null,
            'is_approved'     => $p->is_approved,
            'status'          => $p->status_label, // pending | approved | rejected
            'approved_by'     => $p->approver ? ['id' => $p->approver->id, 'name' => $p->approver->name] : null,
            'approved_at'     => $p->approved_at?->toDateTimeString(),
            'approval_note'   => $p->approval_note,
            'created_at'      => $p->created_at?->toDateTimeString(),
            'user'            => $p->user ? [
                'id'         => $p->user->id,
                'name'       => $p->user->name,
                'email'      => $p->user->email,
                'phone'      => $p->user->phone,
                'position'   => $p->user->position,
                'department' => $p->user->department,
                'image_url'  => $p->user->image_url ? asset($p->user->image_url) : null,
            ] : null,
        ];
    }

    // ============================================================
    // GET /api/company/hr/permissions
    // filter: status, type, user_id, q, date_from, date_to, month, year
    // ============================================================
    public function index(Request $request)
    {
        $this->ensureHr();

        $q = Permission::query()
            ->with([self::USER_COLUMNS, 'approver:id,name'])
            ->where('company_id', $this->companyId());

        $page = $this->applyFilters($q, $request)
            ->orderByRaw('is_approved IS NULL DESC') // pending tampil paling atas
            ->orderByDesc('date_permission')
            ->orderByDesc('id')
            ->paginate((int) $request->query('per_page', 20));

        $page->getCollection()->transform(fn($p) => $this->format($p));

        $base = Permission::where('company_id', $this->companyId());

        return response()->json([
            'status'  => true,
            'message' => 'List permission company',
            'counts'  => [
                'pending'  => (clone $base)->whereNull('is_approved')->count(),
                'approved' => (clone $base)->where('is_approved', true)->count(),
                'rejected' => (clone $base)->where('is_approved', false)->count(),
            ],
            'data' => $page,
        ]);
    }

    // ============================================================
    // GET /api/company/hr/permissions/{id}
    //
    // Sebelumnya detail memakai filter list (status/q). Kalau aplikasi
    // mengirim ?status=pending lalu membuka izin yang sudah approved,
    // hasilnya 404. Sekarang detail hanya dicari berdasarkan id + company.
    // ============================================================
    public function show($id)
    {
        $this->ensureHr();

        $perm = $this->findOrFail($id);

        // Absen karyawan di tanggal izin (untuk dicocokkan HR)
        $start = $perm->date_permission->toDateString();
        $end   = ($perm->end_date ?? $perm->date_permission)->toDateString();

        $attendances = Attendance::where('company_id', $this->companyId())
            ->where('user_id', $perm->user_id)
            ->whereBetween('date', [$start, $end])
            ->orderBy('date')
            ->get()
            ->map(fn($a) => [
                'id'                   => $a->id,
                'date'                 => $a->date?->toDateString(),
                'time_in'              => $a->time_in,
                'time_out'             => $a->time_out,
                'scheduled_in'         => $a->scheduled_in,
                'scheduled_out'        => $a->scheduled_out,
                'status'               => $a->status,
                'late_minutes'         => (int) $a->late_minutes,
                'early_leave_minutes'  => (int) $a->early_leave_minutes,
                'photo_in_url'         => $a->photo_in_url,
                'photo_out_url'        => $a->photo_out_url,
                'latlon_in'            => $a->latlon_in,
                'distance_in_m'        => $a->distance_in_m,
                'is_outside_radius_in' => (bool) $a->is_outside_radius_in,
                'note'                 => $a->note,
            ])->values();

        // Rekap izin karyawan ini di bulan yang sama
        $monthStart = $perm->date_permission->copy()->startOfMonth()->toDateString();
        $monthEnd   = $perm->date_permission->copy()->endOfMonth()->toDateString();

        $monthly = Permission::where('company_id', $this->companyId())
            ->where('user_id', $perm->user_id)
            ->whereBetween('date_permission', [$monthStart, $monthEnd])
            ->selectRaw('type, COUNT(*) as total, SUM(is_approved = 1) as approved')
            ->groupBy('type')
            ->get()
            ->map(fn($r) => [
                'type'       => $r->type,
                'type_label' => Permission::TYPE_LABELS[$r->type] ?? $r->type,
                'total'      => (int) $r->total,
                'approved'   => (int) $r->approved,
            ])->values();

        return response()->json([
            'status'  => true,
            'message' => 'Detail permission',
            'data'    => array_merge($this->format($perm), [
                'attendances'      => $attendances,
                'monthly_summary'  => $monthly,
                'can_approve'      => $perm->is_approved === null,
            ]),
        ]);
    }

    // ============================================================
    // POST /api/company/hr/permissions/{id}/approve
    // body (opsional): { note }
    // ============================================================
    public function approve(Request $request, $id)
    {
        $this->ensureHr();
        $request->validate(['note' => 'nullable|string|max:500']);

        $perm = $this->findOrFail($id);

        if ($perm->is_approved !== null) {
            return response()->json([
                'status'  => false,
                'message' => 'Izin ini sudah ' . ($perm->is_approved ? 'disetujui' : 'ditolak') . ' sebelumnya.',
            ], 422);
        }

        $perm->update([
            'is_approved'   => true,
            'approved_by'   => auth()->id(),
            'approved_at'   => now(),
            'approval_note' => $request->input('note'),
        ]);

        // Koreksi absen yang sudah terlanjur tercatat telat / pulang cepat
        $this->rules->recalculateForPermission($perm);

        // 🔔 Notifikasi ke karyawan
        $this->approvals->decided(ApprovalNotifier::KIND_PERMISSION, $perm, true, $request->input('note'));

        return response()->json([
            'status'  => true,
            'message' => 'Izin disetujui',
            'data'    => $this->format($perm->fresh([self::USER_COLUMNS, 'approver:id,name'])),
        ]);
    }

    // ============================================================
    // POST /api/company/hr/permissions/{id}/reject
    // body: { note } — alasan penolakan wajib
    // ============================================================
    public function reject(Request $request, $id)
    {
        $this->ensureHr();
        $request->validate(
            ['note' => 'required|string|min:3|max:500'],
            ['note.required' => 'Alasan penolakan wajib diisi.']
        );

        $perm = $this->findOrFail($id);

        if ($perm->is_approved !== null) {
            return response()->json([
                'status'  => false,
                'message' => 'Izin ini sudah ' . ($perm->is_approved ? 'disetujui' : 'ditolak') . ' sebelumnya.',
            ], 422);
        }

        $perm->update([
            'is_approved'   => false,
            'approved_by'   => auth()->id(),
            'approved_at'   => now(),
            'approval_note' => $request->input('note'),
        ]);

        $this->rules->recalculateForPermission($perm);

        // 🔔 Notifikasi ke karyawan
        $this->approvals->decided(ApprovalNotifier::KIND_PERMISSION, $perm, false, $request->input('note'));

        return response()->json([
            'status'  => true,
            'message' => 'Izin ditolak',
            'data'    => $this->format($perm->fresh([self::USER_COLUMNS, 'approver:id,name'])),
        ]);
    }

    // ============================================================
    // EXPORT PDF
    // GET /api/company/hr/permissions/export
    // filter sama dengan index
    // ============================================================
    public function export(Request $request)
    {
        $this->ensureHr();

        $q = Permission::query()
            ->with([self::USER_COLUMNS, 'approver:id,name'])
            ->where('company_id', $this->companyId());

        $permissions = $this->applyFilters($q, $request)
            ->orderByDesc('date_permission')
            ->orderByDesc('id')
            ->get();

        $stats = [
            'total'    => $permissions->count(),
            'pending'  => $permissions->whereNull('is_approved')->count(),
            'approved' => $permissions->where('is_approved', true)->count(),
            'rejected' => $permissions->where('is_approved', false)->count(),
        ];

        $bulanLabel = [
            1 => 'Januari',
            2 => 'Februari',
            3 => 'Maret',
            4 => 'April',
            5 => 'Mei',
            6 => 'Juni',
            7 => 'Juli',
            8 => 'Agustus',
            9 => 'September',
            10 => 'Oktober',
            11 => 'November',
            12 => 'Desember',
        ];

        $month       = $request->month;
        $year        = $request->year ?? now()->year;
        $periodLabel = $month
            ? ($bulanLabel[(int) $month] ?? $month) . ' ' . $year
            : ($request->filled('year') ? 'Tahun ' . $year : 'Semua Periode');

        $pdf = \Barryvdh\DomPDF\Facade\Pdf::loadView('pdf.hr_permission', [
            'company'     => auth()->user()->company ?? (object) ['name' => ''],
            'permissions' => $permissions,
            'stats'       => $stats,
            'periodLabel' => $periodLabel,
            'generatedAt' => now()->format('d/m/Y H:i'),
        ])
            ->setPaper('a4', 'portrait')
            ->setOptions(['defaultFont' => 'sans-serif', 'isHtml5ParserEnabled' => true]);

        return $pdf->download('permissions-' . now()->format('Y-m-d') . '.pdf');
    }

<?php

namespace App\Http\Controllers\Api\HrCompany;

use App\Http\Controllers\Controller;
use App\Models\Schedule;
use App\Models\ScheduleParticipant;
use App\Services\Notifiers\ScheduleNotifier;
use Carbon\Carbon;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\Rule;

class HrCompanyScheduleController extends Controller
{
    private const USER_COLS = 'id,name,position,department,image_url';
    private const TYPES     = ['meeting', 'task_duty', 'visit', 'training', 'other'];
    private const STATUSES  = ['upcoming', 'done', 'canceled'];

    public function __construct(private ScheduleNotifier $scheduleNotifier) {}

    // ----------------------------------------------------------
    // HELPERS
    // ----------------------------------------------------------

    // Sebelumnya tidak ada cek role
    private function ensureHr(): void
    {
        if (!auth()->check() || auth()->user()->role !== 'hr') {
            abort(response()->json(['success' => false, 'message' => 'Akses ditolak (khusus HR)'], 403));
        }
    }

    private function companyId(): int
    {
        return (int) auth()->user()->company_id;
    }

    private function relations(): array
    {
        return [
            'user:' . self::USER_COLS,
            'creator:' . self::USER_COLS,
            'participants.user:' . self::USER_COLS,
        ];
    }

    /**
     * Sebelumnya 'exists:users,id' → HR bisa memasukkan user dari COMPANY LAIN.
     * Sekarang user harus satu company.
     */
    private function companyUserRule()
    {
        return Rule::exists('users', 'id')->where('company_id', $this->companyId());
    }

    private function findOrFail(int $id): Schedule
    {
        return Schedule::where('company_id', $this->companyId())->findOrFail($id);
    }

    private const MESSAGES = [
        'start_datetime.required'       => 'Waktu mulai wajib diisi.',
        'start_datetime.after_or_equal' => 'Waktu mulai tidak boleh di masa lalu.',
        'user_id.exists'                => 'Karyawan tidak ditemukan di perusahaan ini.',
        'participants.*.exists'         => 'Ada peserta yang tidak terdaftar di perusahaan ini.',
    ];

    // ============================================================
    // GET /api/company/hr/schedules
    // filter: status, type, user_id, date, start_date, end_date, q, per_page
    // ============================================================
    public function index(Request $request): JsonResponse
    {
        $this->ensureHr();

        $schedules = Schedule::with($this->relations())
            ->where('company_id', $this->companyId())
            ->when($request->status,  fn($q) => $q->where('status', $request->status))
            ->when($request->type,    fn($q) => $q->where('type', $request->type))
            ->when($request->user_id, fn($q) => $q->where('user_id', $request->user_id))
            ->when($request->q,       fn($q) => $q->where('title', 'like', '%' . $request->q . '%'))
            ->filterDates($request)
            ->orderBy('start_datetime', 'asc')
            ->paginate((int) ($request->per_page ?? 15));

        $schedules->getCollection()->transform(fn($s) => $s->toApiArray());

        return response()->json([
            'success' => true,
            'message' => 'Data jadwal berhasil diambil',
            'data'    => $schedules,
        ]);
    }

    // ============================================================
    // POST /api/company/hr/schedules
    //
    // Format waktu yang diterima (semua dianggap / dikonversi ke WIB):
    //   "2026-10-25 09:00"            → 09:00 WIB
    //   "2026-10-25T09:00:00+07:00"   → 09:00 WIB
    //   "2026-10-25T02:00:00.000Z"    → 09:00 WIB (dikonversi dari UTC)
    // ============================================================
    public function store(Request $request): JsonResponse
    {
        $this->ensureHr();

        $request->validate([
            'user_id'             => ['required', $this->companyUserRule()],
            'title'               => ['required', 'string', 'max:255'],
            'description'         => ['nullable', 'string'],
            'type'                => ['required', Rule::in(self::TYPES)],
            'start_datetime'      => ['required', 'date'],
            'end_datetime'        => ['nullable', 'date'],
            'reminder_offsets'    => ['nullable', 'array'],
            'reminder_offsets.*'  => ['integer', 'min:1'],
            'location'            => ['nullable', 'array'],
            'is_recurring'        => ['nullable', 'boolean'],
            'recurrence_type'     => [
                'nullable',
                Rule::requiredIf($request->boolean('is_recurring')),
                Rule::in(['daily', 'weekly', 'monthly']),
            ],
            'recurrence_end_date' => [
                'nullable',
                Rule::requiredIf($request->boolean('is_recurring')),
                'date',
            ],
            'status'              => ['nullable', Rule::in(self::STATUSES)],
            'participants'        => ['nullable', 'array'],
            'participants.*'      => ['integer', $this->companyUserRule()],
        ], self::MESSAGES);

        // Validasi waktu SETELAH dikonversi ke WIB
        // (validasi bawaan "after:now" membandingkan string mentah, bisa meleset 7 jam)
        $start = Carbon::parse(Schedule::toLocal($request->start_datetime));
        $end   = $request->filled('end_datetime') ? Carbon::parse(Schedule::toLocal($request->end_datetime)) : null;

        if ($start->lt(now()->subMinutes(5))) {
            return response()->json([
                'success' => false,
                'message' => 'Waktu mulai tidak boleh di masa lalu.',
            ], 422);
        }

        if ($end && $end->lte($start)) {
            return response()->json([
                'success' => false,
                'message' => 'Waktu selesai harus setelah waktu mulai.',
            ], 422);
        }

        if (
            $request->boolean('is_recurring')
            && Carbon::parse($request->recurrence_end_date)->endOfDay()->lte($start)
        ) {
            return response()->json([
                'success' => false,
                'message' => 'Tanggal akhir pengulangan harus setelah waktu mulai.',
            ], 422);
        }

        DB::beginTransaction();

        try {
            $schedule = Schedule::create([
                'company_id'          => $this->companyId(),
                'user_id'             => $request->user_id,
                'created_by'          => $request->user()->id,
                'title'               => $request->title,
                'description'         => $request->description,
                'type'                => $request->type,
                'start_datetime'      => $request->start_datetime, // dikonversi di model
                'end_datetime'        => $request->end_datetime,
                'reminder_offsets'    => $request->reminder_offsets,
                'location'            => $request->location,
                'is_recurring'        => $request->boolean('is_recurring'),
                'recurrence_type'     => $request->boolean('is_recurring') ? $request->recurrence_type : null,
                'recurrence_end_date' => $request->boolean('is_recurring') ? $request->recurrence_end_date : null,
                'status'              => $request->status ?? 'upcoming',
            ]);

            if ($request->filled('participants')) {
                $rows = collect($request->participants)
                    ->unique()
                    ->reject(fn($uid) => (int) $uid === (int) $request->user_id) // pemilik tidak perlu diundang
                    ->map(fn($uid) => [
                        'schedule_id' => $schedule->id,
                        'user_id'     => $uid,
                        'status'      => 'invited',
                        'created_at'  => now(),
                        'updated_at'  => now(),
                    ])->values()->toArray();

                if ($rows) {
                    ScheduleParticipant::insert($rows);
                }
            }

            DB::commit();

            // 🔔 Notifikasi ke pemilik jadwal + peserta
            $this->scheduleNotifier->created(
                $schedule,
                isset($rows) ? collect($rows)->pluck('user_id')->all() : []
            );
        } catch (\Throwable $e) {
            DB::rollBack();

            return response()->json([
                'success' => false,
                'message' => 'Gagal membuat jadwal',
                'error'   => config('app.debug') ? $e->getMessage() : null,
            ], 500);
        }

        return response()->json([
            'success' => true,
            'message' => 'Jadwal berhasil dibuat',
            'data'    => $schedule->load($this->relations())->toApiArray(),
        ], 201);
    }

    // ============================================================
    // PUT /api/company/hr/schedules/{id}
    // ============================================================
    public function update(Request $request, int $id): JsonResponse
    {
        $this->ensureHr();

        $schedule = $this->findOrFail($id);

        $request->validate([
            'user_id'             => ['sometimes', $this->companyUserRule()],
            'title'               => ['sometimes', 'required', 'string', 'max:255'],
            'description'         => ['nullable', 'string'],
            'type'                => ['sometimes', Rule::in(self::TYPES)],
            'start_datetime'      => ['sometimes', 'required', 'date'],
            'end_datetime'        => ['nullable', 'date'],
            'reminder_offsets'    => ['nullable', 'array'],
            'reminder_offsets.*'  => ['integer', 'min:1'],
            'location'            => ['nullable', 'array'],
            'is_recurring'        => ['sometimes', 'boolean'],
            'recurrence_type'     => ['nullable', Rule::in(['daily', 'weekly', 'monthly'])],
            'recurrence_end_date' => ['nullable', 'date'],
            'status'              => ['sometimes', Rule::in(self::STATUSES)],
        ], self::MESSAGES);

        // Sebelumnya: kirim end_datetime saja → validasi "after:start_datetime"
        // gagal karena start_datetime tidak ikut dikirim. Sekarang dibandingkan
        // dengan nilai lama kalau tidak dikirim.
        $start = $request->has('start_datetime')
            ? Carbon::parse(Schedule::toLocal($request->start_datetime))
            : $schedule->start_datetime;

        $endRaw = $request->has('end_datetime') ? $request->end_datetime : $schedule->end_datetime;
        $end    = $endRaw ? Carbon::parse(Schedule::toLocal($endRaw)) : null;

        if ($end && $start && $end->lte($start)) {
            return response()->json([
                'success' => false,
                'message' => 'Waktu selesai harus setelah waktu mulai.',
            ], 422);
        }

        $data = $request->only([
            'user_id',
            'title',
            'description',
            'type',
            'start_datetime',
            'end_datetime',
            'reminder_offsets',
            'location',
            'is_recurring',
            'recurrence_type',
            'recurrence_end_date',
            'status',
        ]);

        if (array_key_exists('is_recurring', $data) && !$request->boolean('is_recurring')) {
            $data['recurrence_type']     = null;
            $data['recurrence_end_date'] = null;
        }

        // Simpan nilai lama untuk menentukan perlu notifikasi atau tidak
        $old = [
            'title'          => $schedule->title,
            'start_datetime' => $schedule->start_datetime?->format('Y-m-d H:i'),
            'end_datetime'   => $schedule->end_datetime?->format('Y-m-d H:i'),
            'location'       => json_encode($schedule->location),
            'status'         => $schedule->status,
        ];

        $schedule->update($data); // waktu dikonversi ke WIB di model
        $schedule->refresh();

        // 🔔 Notifikasi: dibatalkan, atau judul / waktu / lokasi berubah
        if ($old['status'] !== 'canceled' && $schedule->status === 'canceled') {
            $this->scheduleNotifier->canceled($schedule);
        } elseif (
            $schedule->status === 'upcoming' && (
                $old['title'] !== $schedule->title
                || $old['start_datetime'] !== $schedule->start_datetime?->format('Y-m-d H:i')
                || $old['end_datetime'] !== $schedule->end_datetime?->format('Y-m-d H:i')
                || $old['location'] !== json_encode($schedule->location)
            )
        ) {
            $this->scheduleNotifier->updated($schedule);
        }

        return response()->json([
            'success' => true,
            'message' => 'Jadwal berhasil diupdate',
            'data'    => $schedule->fresh($this->relations())->toApiArray(),
        ]);
    }

    // ============================================================
    // DELETE /api/company/hr/schedules/{id}   (soft delete)
    // ============================================================
    public function destroy(Request $request, int $id): JsonResponse
    {
        $this->ensureHr();

        $schedule = $this->findOrFail($id);

        // 🔔 Notifikasi pembatalan (sebelum dihapus, supaya daftar peserta masih ada)
        if ($schedule->status === 'upcoming' && $schedule->time_status !== 'past') {
            $this->scheduleNotifier->canceled($schedule);
        }

        $schedule->delete();

        return response()->json([
            'success' => true,
            'message' => 'Jadwal berhasil dihapus',
        ]);
    }

    // ============================================================
    // POST /api/company/hr/schedules/{id}/participants
    // body: { participants: [user_id, ...] }
    // ============================================================
    public function addParticipants(Request $request, int $id): JsonResponse
    {
        $this->ensureHr();

        $schedule = $this->findOrFail($id);

        $request->validate([
            'participants'   => ['required', 'array', 'min:1'],
            'participants.*' => ['integer', $this->companyUserRule()],
        ], self::MESSAGES);

        $existing = ScheduleParticipant::where('schedule_id', $schedule->id)
            ->pluck('user_id')
            ->map(fn($v) => (int) $v)
            ->all();

        $added   = [];
        $skipped = [];

        foreach (collect($request->participants)->map(fn($v) => (int) $v)->unique() as $userId) {
            if (in_array($userId, $existing, true) || $userId === (int) $schedule->user_id) {
                $skipped[] = $userId;
                continue;
            }

            ScheduleParticipant::create([
                'schedule_id' => $schedule->id,
                'user_id'     => $userId,
                'status'      => 'invited',
            ]);

            $added[] = $userId;
        }

        // 🔔 Undangan ke peserta yang baru ditambahkan
        if ($added) {
            $this->scheduleNotifier->invited($schedule, $added);
        }

        return response()->json([
            'success' => true,
            'message' => 'Peserta berhasil ditambahkan',
            'data'    => [
                'added'    => $added,
                'skipped'  => $skipped,
                'schedule' => $schedule->fresh($this->relations())->toApiArray(),
            ],
        ]);
    }

        ]);
    }

            ->when($request->type,    fn($q) => $q->where('type', $request->type))

            'canceled' => $schedules->where('status', 'canceled')->count(),

        $pdf = \Barryvdh\DomPDF\Facade\Pdf::loadView('pdf.hr_schedule', [
            'company'     => $request->user()->company ?? (object) ['name' => ''],
            'schedules'   => $schedules,
            'stats'       => $stats,
            'typeLabels'  => Schedule::TYPE_LABELS,
            'generatedAt' => now()->format('d/m/Y H:i'),
        ])
            ->setPaper('a4', 'landscape')
            ->setOptions(['defaultFont' => 'sans-serif', 'isHtml5ParserEnabled' => true]);

        return $pdf->download('schedules-' . now()->format('Y-m-d') . '.pdf');
    }

<?php

namespace App\Http\Controllers\Api\HrCompany;

use App\Http\Controllers\Controller;
use App\Models\Shift;
use App\Models\ShiftGroup;
use App\Models\ShiftGroupAssignment;
use App\Services\Notifiers\ShiftNotifier;
use Illuminate\Http\Request;

class HrCompanyShiftGroupAssignmentController extends Controller
{
    public function __construct(private ShiftNotifier $shiftNotifier) {}

    // Sebelumnya tidak ada cek role
    private function ensureHr(): void
    {
        if (!auth()->check() || auth()->user()->role !== 'hr') {
            abort(response()->json(['success' => false, 'message' => 'Akses ditolak (khusus HR)'], 403));
        }
    }

    private function companyId(): int
    {
        return (int) auth()->user()->company_id;
    }

    private function getGroupOrFail(int $id): ShiftGroup
    {
        return ShiftGroup::where('company_id', $this->companyId())->findOrFail($id);
    }

    private function hasOverlap(int $groupId, string $start, ?string $end, ?int $exceptId = null): bool
    {
        // existing [s1, e1], new [s2, e2] → overlap jika s1 <= e2 && (e1 null || e1 >= s2)
        return ShiftGroupAssignment::where('company_id', $this->companyId())
            ->where('shift_group_id', $groupId)
            ->when($exceptId, fn($q) => $q->where('id', '!=', $exceptId))
            ->where(function ($q) use ($start, $end) {
                if ($end) {
                    $q->where('start_date', '<=', $end);
                }
                $q->where(function ($q2) use ($start) {
                    $q2->whereNull('end_date')->orWhere('end_date', '>=', $start);
                });
            })
            ->exists();
    }

    // ============================================================
    // GET /api/company/hr/shift-groups/{id}/assignments
    // ============================================================
    public function index(Request $request, $groupId)
    {
        $this->ensureHr();

        $group = $this->getGroupOrFail((int) $groupId);

        $data = ShiftGroupAssignment::query()
            ->where('company_id', $this->companyId())
            ->where('shift_group_id', $group->id)
            ->with(['shift:id,name,start_time,end_time'])
            ->orderBy('start_date', 'desc')
            ->paginate((int) $request->get('per_page', 15));

        return response()->json([
            'success' => true,
            'message' => 'List assignment shift group',
            'data'    => $data,
        ]);
    }

    // ============================================================
    // POST /api/company/hr/shift-groups/{id}/assignments
    // body: shift_id, start_date, end_date, note, is_rotation, rotation_cycle_days,
    //       notify (opsional, default true)
    // → notifikasi ke semua anggota grup
    // ============================================================
    public function store(Request $request, $groupId)
    {
        $this->ensureHr();

        $companyId = $this->companyId();
        $group     = $this->getGroupOrFail((int) $groupId);

        $validated = $request->validate([
            'shift_id'            => ['required', 'integer'],
            'start_date'          => ['required', 'date'],
            'end_date'            => ['nullable', 'date', 'after_or_equal:start_date'],
            'note'                => ['nullable', 'string', 'max:500'],
            'is_rotation'         => ['nullable', 'boolean'],
            'rotation_cycle_days' => ['nullable', 'integer', 'min:1'],
            'notify'              => ['nullable', 'boolean'],
        ]);

        $shift = Shift::where('company_id', $companyId)->findOrFail($validated['shift_id']);

        if ($this->hasOverlap($group->id, $validated['start_date'], $validated['end_date'] ?? null)) {
            return response()->json([
                'success' => false,
                'message' => 'Range tanggal bentrok dengan assignment lain untuk group ini.',
            ], 422);
        }

        $assignment = ShiftGroupAssignment::create([
            'company_id'          => $companyId,
            'shift_group_id'      => $group->id,
            'shift_id'            => $shift->id,
            'start_date'          => $validated['start_date'],
            'end_date'            => $validated['end_date'] ?? null,
            'created_by'          => auth()->id(),
            'note'                => $validated['note'] ?? null,
            'is_rotation'         => (bool) ($validated['is_rotation'] ?? false),
            'rotation_cycle_days' => $validated['rotation_cycle_days'] ?? null,
        ]);

        // 🔔 Notifikasi ke anggota grup
        if ($request->boolean('notify', true)) {
            $this->shiftNotifier->groupAssigned($assignment, false);
        }

        $assignment->load(['shift:id,name,start_time,end_time']);

        return response()->json([
            'success' => true,
            'message' => 'Assignment shift group berhasil dibuat.',
            'data'    => $assignment,
        ], 201);
    }

    // ============================================================
    // PUT /api/company/hr/shift-groups/shift-group-assignments/{id}
    // → notifikasi ke anggota grup kalau shift / tanggal berubah
    // ============================================================
    public function update(Request $request, $id)
    {
        $this->ensureHr();

        $companyId  = $this->companyId();
        $assignment = ShiftGroupAssignment::where('company_id', $companyId)->findOrFail((int) $id);

        $validated = $request->validate([
            'shift_id'            => ['required', 'integer'],
            'start_date'          => ['required', 'date'],
            'end_date'            => ['nullable', 'date', 'after_or_equal:start_date'],
            'note'                => ['nullable', 'string', 'max:500'],
            'is_rotation'         => ['nullable', 'boolean'],
            'rotation_cycle_days' => ['nullable', 'integer', 'min:1'],
            'notify'              => ['nullable', 'boolean'],
        ]);

        $shift = Shift::where('company_id', $companyId)->findOrFail($validated['shift_id']);

        if ($this->hasOverlap($assignment->shift_group_id, $validated['start_date'], $validated['end_date'] ?? null, $assignment->id)) {
            return response()->json([
                'success' => false,
                'message' => 'Range tanggal bentrok dengan assignment lain untuk group ini.',
            ], 422);
        }

        $before = [
            'shift_id'   => (int) $assignment->shift_id,
            'start_date' => \Carbon\Carbon::parse($assignment->start_date)->toDateString(),
            'end_date'   => $assignment->end_date ? \Carbon\Carbon::parse($assignment->end_date)->toDateString() : null,
        ];

        $assignment->update([
            'shift_id'            => $shift->id,
            'start_date'          => $validated['start_date'],
            'end_date'            => $validated['end_date'] ?? null,
            'note'                => $validated['note'] ?? null,
            'is_rotation'         => (bool) ($validated['is_rotation'] ?? $assignment->is_rotation),
            'rotation_cycle_days' => $validated['rotation_cycle_days'] ?? $assignment->rotation_cycle_days,
        ]);

        $after = [
            'shift_id'   => (int) $shift->id,
            'start_date' => \Carbon\Carbon::parse($validated['start_date'])->toDateString(),
            'end_date'   => !empty($validated['end_date']) ? \Carbon\Carbon::parse($validated['end_date'])->toDateString() : null,
        ];

namespace App\Http\Controllers\Api\HrCompany;

use App\Http\Controllers\Controller;

class HrCompanyShiftGroupUserController extends Controller

    }

    private function getGroupOrFail(int $id): ShiftGroup
    {
        return ShiftGroup::where('company_id', $this->companyId())->findOrFail($id);
    }

    public function index(Request $request, $id)
    {
        $group = $this->getGroupOrFail((int)$id);

        $users = $group->users()
            ->select('users.id', 'users.name', 'users.email', 'users.department', 'users.position')
            ->orderBy('users.name')
            ->paginate((int)($request->get('per_page', 15)));

        return response()->json([
            'success' => true,
            'message' => 'List user dalam shift group',
            'data' => $users,
        ]);
    }

    /**
     * attach:
     * - Bisa kirim user_ids: [1,2,3]
     * - Atau filter: department, position (bulk)
     * Optional: start_date, end_date untuk masa berlaku membership group
     */
    public function attach(Request $request, $id)
    {
        $companyId = $this->companyId();
        $group = $this->getGroupOrFail((int)$id);

        $validated = $request->validate([
            'user_ids' => ['nullable', 'array'],
            'user_ids.*' => ['integer'],

            'department' => ['nullable', 'string', 'max:100'],
            'position' => ['nullable', 'string', 'max:100'],

            'start_date' => ['nullable', 'date'],
            'end_date' => ['nullable', 'date', 'after_or_equal:start_date'],
        ]);

        $startDate = $validated['start_date'] ?? null;
        $endDate = $validated['end_date'] ?? null;

        $userIds = $validated['user_ids'] ?? [];

        // Bulk by filter jika user_ids kosong
        if (empty($userIds)) {
            $q = User::query()->where('company_id', $companyId);

            if (!empty($validated['department'])) {
                $q->where('department', $validated['department']);
            }
            if (!empty($validated['position'])) {
                $q->where('position', $validated['position']);
            }

            $userIds = $q->pluck('id')->toArray();
        }

        if (empty($userIds)) {
            return response()->json([
                'success' => false,
                'message' => 'Tidak ada user yang dipilih.',
            ], 422);
        }

        // Validasi semua user harus 1 company
        $countValid = User::where('company_id', $companyId)
            ->whereIn('id', $userIds)
            ->count();

        if ($countValid !== count($userIds)) {
            return response()->json([
                'success' => false,
                'message' => 'Ada user yang tidak valid / beda company.',
            ], 422);
        }

        // Attach with pivot data
        $pivotData = [];
        foreach ($userIds as $uid) {
            $pivotData[$uid] = [
                'start_date' => $startDate,
                'end_date' => $endDate,
            ];
        }

        // syncWithoutDetaching biar tidak menghapus anggota lama
        $group->users()->syncWithoutDetaching($pivotData);

        return response()->json([
            'success' => true,
            'message' => 'User berhasil ditambahkan ke shift group.',
            'data' => [
                'shift_group_id' => $group->id,
                'added_count' => count($userIds),
            ],
        ]);
    }

    public function detach(Request $request, $id)
    {
        $group = $this->getGroupOrFail((int)$id);

        $validated = $request->validate([
            'user_ids' => ['required', 'array', 'min:1'],
            'user_ids.*' => ['integer'],
        ]);

        $group->users()->detach($validated['user_ids']);

        return response()->json([
            'success' => true,
            'message' => 'User berhasil dikeluarkan dari shift group.',
            'data' => [
                'shift_group_id' => $group->id,
                'removed_count' => count($validated['user_ids']),
            ],
        ]);
    }
}

class HrCompanyUserShiftOverrideController extends Controller

    public function __construct(private ShiftNotifier $shiftNotifier) {}

    // Sebelumnya tidak ada cek role
    private function ensureHr(): void
    {
        if (!auth()->check() || auth()->user()->role !== 'hr') {

    }

    private function getUserOrFail(int $userId): User
    {
        return User::where('company_id', $this->companyId())->findOrFail($userId);
    }

    // ============================================================
    // GET /api/company/hr/shift-groups/users/{userId}/shift-overrides
    // ============================================================
    public function index(Request $request, $userId)
    {
        $this->ensureHr();

        $user = $this->getUserOrFail((int) $userId);

        $data = UserShiftOverride::query()
            ->where('company_id', $this->companyId())
            ->where('user_id', $user->id)
            ->with(['shift:id,name,start_time,end_time'])
            ->orderBy('start_date', 'desc')
            ->paginate((int) $request->get('per_page', 15));

        return response()->json([
            'success' => true,
            'message' => 'List shift override user',
            'data'    => $data,
        ]);
    }

    // ============================================================
    // POST /api/company/hr/shift-groups/users/{userId}/shift-overrides
    // body: shift_id, start_date, end_date (opsional), reason, notify (opsional, default true)
    // → notifikasi ke karyawan
    // ============================================================
    public function store(Request $request, $userId)
    {
        $this->ensureHr();

        $companyId = $this->companyId();
        $user      = $this->getUserOrFail((int) $userId);

        $validated = $request->validate([
            'shift_id'   => ['required', 'integer'],
            'start_date' => ['required', 'date'],
            'end_date'   => ['nullable', 'date', 'after_or_equal:start_date'],
            'reason'     => ['nullable', 'string', 'max:500'],
            'notify'     => ['nullable', 'boolean'],
        ]);

        $shift = Shift::where('company_id', $companyId)->findOrFail($validated['shift_id']);

        // Cegah override aktif yang bentrok
        $start = $validated['start_date'];
        $end   = $validated['end_date'] ?? null;

        $overlap = UserShiftOverride::where('company_id', $companyId)
            ->where('user_id', $user->id)
            ->where('status', 'active')
            ->where(function ($q) use ($start, $end) {
                if ($end) $q->where('start_date', '<=', $end);
                $q->where(function ($q2) use ($start) {
                    $q2->whereNull('end_date')->orWhere('end_date', '>=', $start);
                });
            })
            ->exists();

        if ($overlap) {
            return response()->json([
                'success' => false,
                'message' => 'Override bentrok dengan override aktif lain untuk user ini.',
            ], 422);
        }

        $override = UserShiftOverride::create([
            'company_id' => $companyId,
            'user_id'    => $user->id,
            'shift_id'   => $shift->id,
            'start_date' => $validated['start_date'],
            'end_date'   => $validated['end_date'] ?? null,
            'status'     => 'active',
            'created_by' => auth()->id(),
            'reason'     => $validated['reason'] ?? null,
        ]);

        // 🔔 Notifikasi ke karyawan
        if ($request->boolean('notify', true)) {
            $this->shiftNotifier->overrideCreated($override);
        }

        $override->load(['shift:id,name,start_time,end_time']);

        return response()->json([
            'success' => true,
            'message' => 'Override shift berhasil dibuat.',
            'data'    => $override,
        ], 201);
    }

    // ============================================================
    // PATCH /api/company/hr/shift-groups/user-shift-overrides/{id}/cancel
    // → notifikasi ke karyawan
    // ============================================================
    public function cancel($id)
    {
        $this->ensureHr();

        $override = UserShiftOverride::where('company_id', $this->companyId())->findOrFail((int) $id);

        $wasActive = $override->status === 'active';

        $override->update(['status' => 'cancelled']);

        // 🔔 Notifikasi ke karyawan (hanya kalau sebelumnya aktif & belum lewat)
        if ($wasActive && (!$override->end_date || now()->startOfDay()->lte($override->end_date))) {
            $this->shiftNotifier->overrideCanceled($override);
        }

        return response()->json([
            'success' => true,
            'message' => 'Override shift berhasil dibatalkan.',
            'data'    => $override,
        ]);
    }

    // ============================================================
    // DELETE /api/company/hr/shift-groups/user-shift-overrides/{id}
    // → notifikasi ke karyawan kalau override masih aktif
    // ============================================================
    public function destroy($id)
    {
        $this->ensureHr();

        $override = UserShiftOverride::where('company_id', $this->companyId())->findOrFail((int) $id);

        $notify = $override->status === 'active'
            && (!$override->end_date || now()->startOfDay()->lte($override->end_date));

        $snapshot = $override->replicate();
        $snapshot->id = $override->id;

        $override->delete();

        if ($notify) {
            $this->shiftNotifier->overrideCanceled($snapshot);
        }

        return response()->json([
            'success' => true,
            'message' => 'Override shift berhasil dihapus.',
        ]);
    }
}

    public function getPeriodLabelAttribute(): string
    {
        $s = $this->start_date;

        'date_permission' => 'date:Y-m-d',
        'end_date'        => 'date:Y-m-d',
        'is_approved'     => 'boolean',
        'approved_at'     => 'datetime',

    // RELATIONS
    // ----------------------------------------------------------

    {
        return $this->belongsTo(User::class);
    }

    {
        return $this->belongsTo(Company::class);
    }

    public function approver()
    {
        return $this->belongsTo(User::class, 'approved_by');
    }

    // ----------------------------------------------------------
    // SCOPES
    // ----------------------------------------------------------

    public function scopeApproved(Builder $query): Builder
    {
        return $query->where('is_approved', true);
    }

    /**
     * Izin yang berlaku pada tanggal tertentu
     * (date_permission <= date <= COALESCE(end_date, date_permission)).
     */
    public function scopeActiveOn(Builder $query, string $date): Builder
    {
        return $query
            ->whereDate('date_permission', '<=', $date)
            ->where(function ($q) use ($date) {
                $q->where(function ($q2) use ($date) {
                    $q2->whereNull('end_date')->whereDate('date_permission', $date);
                })->orWhereDate('end_date', '>=', $date);
            });
    }

            false   => 'rejected',
            default => 'pending',
        };
    }

    {
        return self::TYPE_LABELS[$this->type ?? self::TYPE_TIDAK_MASUK] ?? 'Izin';

    public function scopeFilterDates($query, $request)

    /** upcoming | ongoing | past | done | canceled — dihitung dari jam sekarang */
    public function getTimeStatusAttribute(): string
    {
        if (in_array($this->status, ['done', 'canceled'], true)) {
            return $this->status;
        }

        $now   = now();
        $start = $this->start_datetime;
        $end   = $this->end_datetime ?? $start?->copy()->addHour();

        if (!$start)            return 'upcoming';
        if ($now->lt($start))   return 'upcoming';
        if ($now->lte($end))    return 'ongoing';

        return 'past';
    }

                ? self::DAY_NAMES[$start->dayOfWeek] . ', ' . $start->day . ' '
                . self::MONTH_NAMES[$start->month] . ' ' . $start->year
                : null,
            'start_time'          => $start?->format('H:i'),
            'end_time'            => $end?->format('H:i'),
            'time_label'          => $start
                ? $start->format('H:i') . ($end ? ' – ' . $end->format('H:i') : '') . ' WIB'
                : null,
            'duration_minutes'    => $start && $end ? (int) $start->diffInMinutes($end, true) : null,
            'timezone'            => config('app.timezone'),

            'time_status'         => $this->time_status,
            'created_at'          => self::fmt($this->created_at),
            'updated_at'          => self::fmt($this->updated_at),
        ];

        if ($this->relationLoaded('user')) {
            $data['user'] = self::userBrief($this->user);
        }

        if ($this->relationLoaded('creator')) {
            $data['creator'] = self::userBrief($this->creator);
        }

        if ($this->relationLoaded('participants')) {
            $participants = $this->participants;

            $data['participants'] = $participants->map(fn($p) => [
                'id'           => $p->id,
                'user_id'      => $p->user_id,
                'status'       => $p->status,
                'note'         => $p->note,
                'responded_at' => self::fmt($p->responded_at),
                'user'         => $p->relationLoaded('user') ? self::userBrief($p->user) : null,
            ])->values();

            $data['participant_counts'] = [
                'total'    => $participants->count(),
                'invited'  => $participants->where('status', 'invited')->count(),
                'accepted' => $participants->where('status', 'accepted')->count(),
                'declined' => $participants->where('status', 'declined')->count(),
            ];

            if ($viewerId !== null) {
                $mine = $participants->firstWhere('user_id', $viewerId);
                $data['my_participation'] = $mine ? [
                    'id'           => $mine->id,
                    'status'       => $mine->status,
                    'note'         => $mine->note,
                    'responded_at' => self::fmt($mine->responded_at),
                ] : null;
            }
        }

        if ($viewerId !== null) {
            $data['is_owner'] = (int) $this->user_id === $viewerId;
        }

        return $data;
    }

<?php

namespace App\Services;

use Illuminate\Support\Facades\File;
use Illuminate\Support\Str;
use Illuminate\Validation\ValidationException;

/**
 * Menyimpan foto yang dikirim aplikasi dalam bentuk STRING base64.
 *
 * Aplikasi mobile cukup kirim:
 *   "photo": "/9j/4AAQSkZJRgABAQ..."              (base64 polos)
 *   "photo": "data:image/jpeg;base64,/9j/4AAQ..."  (data URI juga diterima)
 *
 * Request tidak perlu multipart/form-data. Di database hanya tersimpan
 * path pendek seperti "image/attendance/2026/10/9b1f...jpg".
 */
class AttendancePhotoService
{
    private const MAX_BYTES = 3 * 1024 * 1024; // 3 MB setelah di-decode

    private const ALLOWED_MIMES = [
        'image/jpeg' => 'jpg',
        'image/png'  => 'png',
        'image/webp' => 'webp',
    ];

    public function storeBase64(string $base64, string $field = 'photo', string $folder = 'image/attendance'): string
    {
        // Buang prefix data URI kalau ada
        if (str_contains($base64, ',') && str_starts_with(trim($base64), 'data:')) {
            $base64 = substr($base64, strpos($base64, ',') + 1);
        }

        $base64 = preg_replace('/\s+/', '', $base64);
        $binary = base64_decode($base64, true);

        if ($binary === false || $binary === '') {
            throw ValidationException::withMessages([
                $field => 'Format foto tidak valid (harus base64).',
            ]);
        }

        if (strlen($binary) > self::MAX_BYTES) {
            throw ValidationException::withMessages([
                $field => 'Ukuran foto maksimal 3 MB. Kompres foto di aplikasi sebelum dikirim.',
            ]);
        }

        $mime = (new \finfo(FILEINFO_MIME_TYPE))->buffer($binary);

        if (!isset(self::ALLOWED_MIMES[$mime])) {
            throw ValidationException::withMessages([
                $field => 'Foto harus berformat JPG, PNG, atau WEBP.',
            ]);
        }

        $relativeDir = trim($folder, '/') . '/' . now()->format('Y/m');
        $absoluteDir = public_path($relativeDir);

        if (!File::exists($absoluteDir)) {
            File::makeDirectory($absoluteDir, 0755, true);
        }

        $fileName = Str::uuid()->toString() . '.' . self::ALLOWED_MIMES[$mime];
        File::put($absoluteDir . DIRECTORY_SEPARATOR . $fileName, $binary);

        return $relativeDir . '/' . $fileName;
    }

    public function delete(?string $relativePath): void
    {
        if ($relativePath && File::exists(public_path($relativePath))) {
            File::delete(public_path($relativePath));
        }
    }
}

<?php

namespace App\Services;

use App\Models\Attendance;
use App\Models\Company;
use App\Models\Permission;
use App\Models\Shift;
use App\Models\User;
use Carbon\Carbon;
use Illuminate\Support\Collection;
use Illuminate\Support\Facades\DB;

/**
 * Semua aturan absen karyawan dikumpulkan di sini supaya dipakai sama persis oleh:
 *  - EmployeeAttendanceController (check-in / check-out)
 *  - HrCompanyPermissionController (hitung ulang saat izin di-approve / reject)
 *
 * Aturan izin:
 *  - Izin hanya berpengaruh kalau SUDAH di-approve HR (is_approved = true).
 *  - datang_terlambat s/d 10:00, datang 10:00  → tidak telat.
 *  - datang_terlambat s/d 10:00, datang 11:00  → telat 60 menit (dihitung dari 10:00).
 *  - keluar_kantor yang dimulai sejak jam masuk diperlakukan sama seperti datang_terlambat.
 *  - pulang_cepat mulai 14:00 → pulang 14:00 tidak dihitung pulang cepat.
 *  - dinas_luar → radius tidak dicek, lokasi tetap dicatat.
 */
class AttendanceRuleService
{
    // ----------------------------------------------------------
    // SHIFT
    // Prioritas: override → group → default → null (pakai jam company)
    // ----------------------------------------------------------

    public function resolveShift(User $user, string $date): ?Shift
    {
        $override = DB::table('user_shift_overrides')
            ->where('user_id', $user->id)
            ->where('status', 'active')
            ->where('start_date', '<=', $date)
            ->where(function ($q) use ($date) {
                $q->whereNull('end_date')->orWhere('end_date', '>=', $date);
            })
            ->whereNull('deleted_at')
            ->orderByDesc('start_date')
            ->first();

        if ($override) {
            return Shift::find($override->shift_id);
        }

        $groupIds = DB::table('shift_group_users')
            ->where('user_id', $user->id)
            ->where(function ($q) use ($date) {
                $q->whereNull('start_date')->orWhere('start_date', '<=', $date);
            })
            ->where(function ($q) use ($date) {
                $q->whereNull('end_date')->orWhere('end_date', '>=', $date);
            })
            ->pluck('shift_group_id');

        if ($groupIds->isNotEmpty()) {
            $assignment = DB::table('shift_group_assignments')
                ->whereIn('shift_group_id', $groupIds)
                ->where('company_id', $user->company_id)
                ->where('start_date', '<=', $date)
                ->where(function ($q) use ($date) {
                    $q->whereNull('end_date')->orWhere('end_date', '>=', $date);
                })
                ->whereNull('deleted_at')
                ->orderByDesc('start_date')
                ->first();

            if ($assignment) {
                return Shift::find($assignment->shift_id);
            }
        }

        return Shift::where('company_id', $user->company_id)
            ->where('is_default', true)
            ->first();
    }

    public function scheduledIn(?Shift $shift, Company $company, string $date): ?Carbon
    {
        $time = $shift ? $shift->start_time : $company->time_in;

        return $time ? Carbon::parse($date . ' ' . $time) : null;
    }

    public function scheduledOut(?Shift $shift, Company $company, string $date): ?Carbon
    {
        $endTime = $shift ? $shift->end_time : $company->time_out;
        if (!$endTime) {
            return null;
        }

        $out = Carbon::parse($date . ' ' . $endTime);

        // Shift malam (misal 22:00 - 06:00): jam pulang jatuh di hari berikutnya
        $in = $this->scheduledIn($shift, $company, $date);
        if ($in && $out->lte($in)) {
            $out->addDay();
        }

        return $out;
    }

    // ----------------------------------------------------------
    // IZIN
    // ----------------------------------------------------------

    /** Semua izin yang SUDAH di-approve dan berlaku di tanggal tsb. */
    public function approvedPermissionsOn(int $userId, int $companyId, string $date): Collection
    {
        return Permission::query()
            ->where('user_id', $userId)
            ->where('company_id', $companyId)
            ->approved()
            ->activeOn($date)
            ->orderByDesc('id')
            ->get();
    }

    public function findDinasLuar(Collection $permissions): ?Permission
    {
        return $permissions->firstWhere('type', Permission::TYPE_DINAS_LUAR);
    }

    /**
     * Jam masuk yang diizinkan. Mengembalikan [Carbon|null, Permission|null].
     */
    public function allowedIn(?Carbon $scheduledIn, Collection $permissions, string $date): array
    {
        if (!$scheduledIn) {
            return [null, null];
        }

        $best = null;
        $bestTime = $scheduledIn->copy();

        foreach ($permissions as $perm) {
            $candidate = null;

            if ($perm->type === Permission::TYPE_DATANG_TERLAMBAT && $perm->end_time) {
                $candidate = Carbon::parse($date . ' ' . $perm->end_time);
            }

            // Izin keluar yang dimulai saat/ sebelum jam masuk = datang terlambat
            if (
                $perm->type === Permission::TYPE_KELUAR_KANTOR
                && $perm->start_time && $perm->end_time
                && Carbon::parse($date . ' ' . $perm->start_time)->lte($scheduledIn)
            ) {
                $candidate = Carbon::parse($date . ' ' . $perm->end_time);
            }

            if ($candidate && $candidate->gt($bestTime)) {
                $bestTime = $candidate;
                $best = $perm;
            }
        }

        return [$bestTime, $best];
    }

    /**
     * Jam pulang yang diizinkan. Mengembalikan [Carbon|null, Permission|null].
     */
    public function allowedOut(?Carbon $scheduledOut, Collection $permissions, string $date): array
    {
        if (!$scheduledOut) {
            return [null, null];
        }

        $best = null;
        $bestTime = $scheduledOut->copy();

        foreach ($permissions as $perm) {
            $candidate = null;

            if ($perm->type === Permission::TYPE_PULANG_CEPAT && $perm->start_time) {
                $candidate = Carbon::parse($date . ' ' . $perm->start_time);
            }

            // Izin keluar yang berlangsung sampai jam pulang = pulang cepat
            if (
                $perm->type === Permission::TYPE_KELUAR_KANTOR
                && $perm->start_time && $perm->end_time
                && Carbon::parse($date . ' ' . $perm->end_time)->gte($scheduledOut)
            ) {
                $candidate = Carbon::parse($date . ' ' . $perm->start_time);
            }

            if ($candidate && $candidate->lt($bestTime)) {
                $bestTime = $candidate;
                $best = $perm;
            }
        }

        return [$bestTime, $best];
    }

    // ----------------------------------------------------------
    // HITUNG TELAT & PULANG CEPAT
    // ----------------------------------------------------------

    /**
     * @return array{late_minutes:int, permission:?Permission}
     */
    public function calcLate(
        ?Shift $shift,
        Company $company,
        Carbon $actualIn,
        string $date,
        Collection $permissions
    ): array {
        $scheduledIn = $this->scheduledIn($shift, $company, $date);
        if (!$scheduledIn) {
            return ['late_minutes' => 0, 'permission' => null];
        }

        [$allowedIn, $usedPermission] = $this->allowedIn($scheduledIn, $permissions, $date);

        // Grace period hanya berlaku untuk jadwal normal.
        // Kalau pakai izin, jam izin itu sendiri sudah jadi kelonggarannya.
        $deadline = $usedPermission
            ? $allowedIn->copy()
            : $allowedIn->copy()->addMinutes($shift?->grace_period_minutes ?? 0);

        if ($actualIn->lte($deadline)) {
            return ['late_minutes' => 0, 'permission' => $usedPermission];
        }

        // diffInMinutes($x, true) = selalu positif, aman untuk Carbon 2 & 3
        return [
            'late_minutes' => (int) $allowedIn->diffInMinutes($actualIn, true),
            'permission'   => $usedPermission,
        ];
    }

    /**
     * @return array{early_leave_minutes:int, permission:?Permission}
     */
    public function calcEarlyLeave(
        ?Shift $shift,
        Company $company,
        Carbon $actualOut,
        string $date,
        Collection $permissions
    ): array {
        $scheduledOut = $this->scheduledOut($shift, $company, $date);
        if (!$scheduledOut) {
            return ['early_leave_minutes' => 0, 'permission' => null];
        }

        [$allowedOut, $usedPermission] = $this->allowedOut($scheduledOut, $permissions, $date);

        if ($actualOut->gte($allowedOut)) {
            return ['early_leave_minutes' => 0, 'permission' => $usedPermission];
        }

        return [
            'early_leave_minutes' => (int) $allowedOut->diffInMinutes($actualOut, true),
            'permission'          => $usedPermission,
        ];
    }

    public function buildNote(?Permission $permission): ?string
    {
        if (!$permission) {
            return null;
        }

        $label = Permission::TYPE_LABELS[$permission->type] ?? 'Izin';

        $range = match (true) {
            $permission->start_time && $permission->end_time =>
            ' ' . substr($permission->start_time, 0, 5) . '–' . substr($permission->end_time, 0, 5),
            (bool) $permission->end_time   => ' s/d ' . substr($permission->end_time, 0, 5),
            (bool) $permission->start_time => ' mulai ' . substr($permission->start_time, 0, 5),
            default                        => '',
        };

        return $label . $range . ' — ' . $permission->reason;
    }

    // ----------------------------------------------------------
    // HITUNG ULANG
    // Dipanggil saat HR approve / reject izin SETELAH karyawan sudah absen.
    // ----------------------------------------------------------

    public function recalculate(int $userId, int $companyId, string $date): ?Attendance
    {
        $attendance = Attendance::where('company_id', $companyId)
            ->where('user_id', $userId)
            ->whereDate('date', $date)
            ->first();

        // Absen manual oleh HR (alpa, izin seharian, dll) tidak diubah otomatis
        if (!$attendance || !$attendance->time_in || $attendance->marked_by) {
            return $attendance;
        }

        $user    = User::findOrFail($userId);
        $company = Company::findOrFail($companyId);
        $shift   = $attendance->shift_id ? Shift::find($attendance->shift_id) : $this->resolveShift($user, $date);
        $perms   = $this->approvedPermissionsOn($userId, $companyId, $date);

        $late = $this->calcLate($shift, $company, Carbon::parse($date . ' ' . $attendance->time_in), $date, $perms);

        $early = ['early_leave_minutes' => 0, 'permission' => null];
        if ($attendance->time_out) {
            $out = Carbon::parse($date . ' ' . $attendance->time_out);
            $scheduledIn = $this->scheduledIn($shift, $company, $date);
            if ($scheduledIn && $out->lt($scheduledIn)) {
                $out->addDay(); // pulang shift malam
            }
            $early = $this->calcEarlyLeave($shift, $company, $out, $date, $perms);
        }

        $dinasLuar = $this->findDinasLuar($perms);
        $used = $late['permission'] ?? $early['permission'] ?? $dinasLuar;

        $attendance->late_minutes        = $late['late_minutes'];
        $attendance->early_leave_minutes = $early['early_leave_minutes'];
        $attendance->status              = $late['late_minutes'] > 0 ? 'late' : 'on_time';
        $attendance->permission_id       = $used?->id;
        $attendance->note                = $this->buildNote($used);
        $attendance->save();

        return $attendance;
    }

    /**
     * Hitung ulang semua tanggal yang dicakup sebuah izin.
     */
    public function recalculateForPermission(Permission $permission): void
    {
        $start = Carbon::parse($permission->date_permission)->startOfDay();
        $end   = $permission->end_date
            ? Carbon::parse($permission->end_date)->startOfDay()
            : $start->copy();

        for ($day = $start->copy(); $day->lte($end); $day->addDay()) {
            $this->recalculate($permission->user_id, $permission->company_id, $day->toDateString());
        }
    }

    // ----------------------------------------------------------
    // LOKASI
    // ----------------------------------------------------------

    public function distanceMeters(Company $company, float $lat, float $lng): int
    {
        $R    = 6371000;
        $lat1 = (float) $company->latitude;
        $lng1 = (float) $company->longitude;
        $dLat = deg2rad($lat - $lat1);
        $dLng = deg2rad($lng - $lng1);
        $a    = sin($dLat / 2) ** 2
            + cos(deg2rad($lat1)) * cos(deg2rad($lat)) * sin($dLng / 2) ** 2;

        return (int) round($R * 2 * atan2(sqrt($a), sqrt(1 - $a)));
    }

    public function radiusMeters(Company $company): int
    {
        return (int) round(((float) $company->radius_km) * 1000);
    }
}

<?php

namespace App\Services;

use App\Models\CompanyHoliday;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Str;

/**
 * Notifikasi libur ke HP karyawan.
 *  - created  : HR menambah libur
 *  - updated  : HR mengubah tanggal / nama libur
 *  - canceled : HR menghapus libur yang belum lewat
 *  - reminder : H-1 sebelum libur (dijalankan scheduler)
 *
 * Versi baru: lewat NotificationService → selain pop-up,
 * notifikasi libur sekarang juga masuk kotak masuk aplikasi.
 */
class HolidayNotifier
{
    public function __construct(private NotificationService $notif) {}

    /**
     * @return array{sent:int, failed:int, skipped:int, stored:int}
     */
    private function send(CompanyHoliday $h, string $event, string $title, string $body): array
    {
        $r = $this->notif->send(
            $this->notif->employeesOf((int) $h->company_id),
            'holiday.' . $event,
            $title,
            $body,
            [
                'category'       => 'holiday',
                'action'         => 'holiday_detail',
                // Untuk "canceled" data libur sudah terhapus, jadi pakai id manual
                'reference_type' => 'CompanyHoliday',
                'reference_id'   => $h->id,
                'data'           => [
                    'holiday_id' => $h->id,
                    'event'      => $event,
                    'start_date' => $h->start_date?->format('Y-m-d'),
                    'end_date'   => $h->end_date?->format('Y-m-d'),
                ],
            ]
        );

        $result = [
            'sent'    => $r['sent'],
            'failed'  => $r['failed'],
            'skipped' => $r['no_token'],
            'stored'  => $r['stored'],
        ];

        Log::info("Notifikasi libur [{$event}]", ['holiday_id' => $h->id] + $result);

        return $result;
    }

    public function created(CompanyHoliday $h): array
    {
        return $this->send($h, 'created', '📅 Libur: ' . $h->name, $h->period_label . $this->noteSuffix($h));
    }

    public function updated(CompanyHoliday $h): array
    {
        return $this->send($h, 'updated', '📅 Jadwal libur diperbarui', $h->name . ' — ' . $h->period_label);
    }

    public function canceled(CompanyHoliday $h): array
    {
        return $this->send($h, 'canceled', '❌ Libur dibatalkan', $h->name . ' (' . $h->period_label . ') dibatalkan. Masuk kerja seperti biasa.');
    }

    public function reminder(CompanyHoliday $h): array
    {
        return $this->send($h, 'reminder', '🔔 Besok libur: ' . $h->name, $h->period_label . $this->noteSuffix($h));
    }

    private function noteSuffix(CompanyHoliday $h): string
    {
        return $h->note ? ' — ' . Str::limit($h->note, 80) : '';
    }
}

===== app\Services\Notifiers\ShiftNotifier.php =====
<?php

namespace App\Services\Notifiers;

use App\Models\Shift;
use App\Services\NotificationService;
use Carbon\Carbon;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;

/**
 * Notifikasi perubahan shift oleh HR.
 *
 *   overrideCreated()   → 1 karyawan: shift khusus untuk dia
 *   overrideCanceled()  → 1 karyawan: shift khusus dibatalkan, kembali ke shift biasa
 *   groupAssigned()     → semua anggota grup: grup dapat shift baru / diubah
 */
class ShiftNotifier
{
    private const BULAN = [
        1 => 'Jan',
        'Feb',
        'Mar',
        'Apr',
        'Mei',
        'Jun',
        'Jul',
        'Agu',
        'Sep',
        'Okt',
        'Nov',
        'Des',
    ];

    public function __construct(private NotificationService $notif) {}

    private function tgl($date): string
    {
        $d = Carbon::parse($date);

        return $d->day . ' ' . self::BULAN[$d->month] . ' ' . $d->year;
    }

    private function period($start, $end): string
    {
        if (!$end) {
            return 'mulai ' . $this->tgl($start);
        }

        if (Carbon::parse($start)->isSameDay(Carbon::parse($end))) {
            return 'tanggal ' . $this->tgl($start);
        }

        return $this->tgl($start) . ' – ' . $this->tgl($end);
    }

    private function shiftLabel(?Shift $shift): string
    {
        if (!$shift) return 'shift baru';

        return $shift->name . ' (' . substr($shift->start_time, 0, 5) . '–' . substr($shift->end_time, 0, 5) . ')';
    }

    private function safe(callable $fn, string $event): void
    {
        try {
            $fn();
        } catch (\Throwable $e) {
            Log::error('Notif shift gagal', ['event' => $event, 'error' => $e->getMessage()]);
        }
    }

    // ----------------------------------------------------------

    public function overrideCreated($override): void
    {
        $this->safe(function () use ($override) {
            $shift = Shift::find($override->shift_id);

            $this->notif->sendLater(
                $override->user_id,
                'shift.override_created',
                '🔄 Shift Anda diubah',
                'Shift ' . $this->shiftLabel($shift) . ' ' . $this->period($override->start_date, $override->end_date)
                    . ($override->reason ? '. Alasan: ' . $override->reason : ''),
                [
                    'category'  => 'schedule',
                    'action'    => 'shift_schedule',
                    'reference' => $override,
                    'data'      => [
                        'shift_id'   => $override->shift_id,
                        'start_date' => Carbon::parse($override->start_date)->toDateString(),
                    ],
                ]
            );
        }, 'override_created');
    }

    public function overrideCanceled($override): void
    {
        $this->safe(function () use ($override) {
            $shift = Shift::find($override->shift_id);

            $this->notif->sendLater(
                $override->user_id,
                'shift.override_canceled',
                '↩️ Perubahan shift dibatalkan',
                'Shift ' . $this->shiftLabel($shift) . ' ' . $this->period($override->start_date, $override->end_date)
                    . ' dibatalkan. Anda kembali ke shift biasa.',
                [
                    'category'  => 'schedule',
                    'action'    => 'shift_schedule',
                    'reference' => $override,
                    'data'      => [
                        'start_date' => Carbon::parse($override->start_date)->toDateString(),
                    ],
                ]
            );
        }, 'override_canceled');
    }

    /** Anggota grup yang aktif di periode assignment */
    private function groupMemberIds(int $groupId, $start, $end): array
    {
        return DB::table('shift_group_users')
            ->where('shift_group_id', $groupId)
            // keanggotaan sudah dimulai sebelum assignment berakhir
            ->when($end, fn($q) => $q->where(function ($q2) use ($end) {
                $q2->whereNull('start_date')->orWhere('start_date', '<=', $end);
            }))
            // dan belum berakhir saat assignment dimulai
            ->where(function ($q) use ($start) {
                $q->whereNull('end_date')->orWhere('end_date', '>=', $start);
            })
            ->pluck('user_id')
            ->map(fn($v) => (int) $v)
            ->unique()
            ->values()
            ->all();
    }

    public function groupAssigned($assignment, bool $isUpdate = false): void
    {
        $this->safe(function () use ($assignment, $isUpdate) {
            $userIds = $this->groupMemberIds(
                (int) $assignment->shift_group_id,
                Carbon::parse($assignment->start_date)->toDateString(),
                $assignment->end_date ? Carbon::parse($assignment->end_date)->toDateString() : null
            );

            if (!$userIds) return;

            $shift = Shift::find($assignment->shift_id);
            $group = DB::table('shift_groups')->where('id', $assignment->shift_group_id)->value('name');

            $this->notif->sendLater(
                $userIds,
                $isUpdate ? 'shift.group_updated' : 'shift.group_assigned',
                $isUpdate ? '🔄 Jadwal shift grup diubah' : '📋 Jadwal shift baru',
                'Grup ' . ($group ?: '-') . ': ' . $this->shiftLabel($shift) . ' '
                    . $this->period($assignment->start_date, $assignment->end_date)
                    . ($assignment->note ? '. ' . $assignment->note : ''),
                [
                    'category'  => 'schedule',
                    'action'    => 'shift_schedule',
                    'reference' => $assignment,
                    'data'      => [
                        'shift_id'       => $assignment->shift_id,
                        'shift_group_id' => $assignment->shift_group_id,
                        'start_date'     => Carbon::parse($assignment->start_date)->toDateString(),
                    ],
                ]
            );
        }, 'group_assigned');
    }
}
